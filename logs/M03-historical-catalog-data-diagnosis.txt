    s.code AS season,
    COUNT(DISTINCT p.id) AS currently_catalog_eligible
FROM players p
JOIN player_reconciliation_results rec
    ON rec.player_id = p.id
JOIN seasons s
    ON s.id = rec.season_id
JOIN player_registrations registration
    ON registration.player_id = p.id
   AND registration.season_id = s.id
JOIN clubs c
    ON c.id = registration.club_id
JOIN season_clubs sc
    ON sc.season_id = s.id
   AND sc.club_id = c.id
JOIN player_roles role_fantacalcio
    ON role_fantacalcio.player_id = p.id
   AND role_fantacalcio.season_id = s.id
   AND role_fantacalcio.source = 'FANTACALCIO'
WHERE p.active = TRUE
  AND registration.registration_status = 'ACTIVE'
  AND role_fantacalcio.fantasy_role IS NOT NULL
