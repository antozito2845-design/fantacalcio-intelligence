WITH historical AS (
    SELECT DISTINCT stats.player_id
    FROM player_season_statistics stats
    JOIN seasons s ON s.id = stats.season_id
    WHERE s.current_season = FALSE

    UNION

    SELECT DISTINCT r.player_id
    FROM player_roles r
    JOIN seasons s ON s.id = r.season_id
    WHERE s.current_season = FALSE
),
current_catalog AS (
    SELECT DISTINCT p.id AS player_id
    FROM players p
    JOIN player_reconciliation_results rec
        ON rec.player_id = p.id
    JOIN seasons s
        ON s.id = rec.season_id
       AND s.current_season = TRUE
    JOIN player_registrations registration
        ON registration.player_id = p.id
       AND registration.season_id = s.id
    JOIN season_clubs sc
        ON sc.season_id = s.id
       AND sc.club_id = registration.club_id
    JOIN player_roles role
        ON role.player_id = p.id
       AND role.season_id = s.id
       AND role.source = 'FANTACALCIO'
    WHERE p.active = TRUE
      AND registration.registration_status = 'ACTIVE'
      AND role.fantasy_role IS NOT NULL
)
SELECT
    COUNT(*) AS historical_identities,
    COUNT(*) FILTER (
        WHERE current.player_id IS NULL
    ) AS historical_not_current_catalog,
    COUNT(*) FILTER (
        WHERE current.player_id IS NOT NULL
    ) AS historical_still_current
FROM historical h
LEFT JOIN current_catalog current
    ON current.player_id = h.player_id;

WITH role_players AS (
    SELECT DISTINCT
        r.player_id,
        r.season_id
    FROM player_roles r
    WHERE r.source = 'FANTACALCIO'
),
stat_players AS (
    SELECT DISTINCT
        player_id,
        season_id
    FROM player_season_statistics
),
all_pairs AS (
    SELECT player_id, season_id FROM role_players
    UNION
    SELECT player_id, season_id FROM stat_players
)
SELECT
    s.code AS season,
    COUNT(DISTINCT ap.player_id) AS historical_identities,
    COUNT(DISTINCT rp.player_id) AS with_fantacalcio_role,
    COUNT(DISTINCT sp.player_id) AS with_statistics,
    COUNT(DISTINCT pr.player_id) AS with_registration
FROM seasons s
LEFT JOIN all_pairs ap
    ON ap.season_id = s.id
LEFT JOIN role_players rp
    ON rp.season_id = s.id
   AND rp.player_id = ap.player_id
LEFT JOIN stat_players sp
    ON sp.season_id = s.id
   AND sp.player_id = ap.player_id
LEFT JOIN player_registrations pr
    ON pr.season_id = s.id
   AND pr.player_id = ap.player_id
GROUP BY s.id, s.code, s.start_year
ORDER BY s.start_year;
