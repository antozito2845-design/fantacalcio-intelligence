WITH historical AS (
    SELECT DISTINCT stats.player_id
    FROM player_season_statistics stats
    JOIN seasons s
        ON s.id = stats.season_id
    WHERE s.current_season = FALSE

    UNION

    SELECT DISTINCT r.player_id
    FROM player_roles r
    JOIN seasons s
        ON s.id = r.season_id
    WHERE s.current_season = FALSE
),
current_catalog AS (
    SELECT DISTINCT p.id AS player_id
    FROM players p
    JOIN player_reconciliation_results rec
        ON rec.player_id = p.id
    JOIN seasons s
        ON s.id = rec.season_id
       AND s.current_season = TRUE
    JOIN player_registrations registration
        ON registration.player_id = p.id
       AND registration.season_id = s.id
    JOIN season_clubs sc
        ON sc.season_id = s.id
       AND sc.club_id = registration.club_id
    JOIN player_roles role
        ON role.player_id = p.id
       AND role.season_id = s.id
       AND role.source = 'FANTACALCIO'
    WHERE p.active = TRUE
      AND registration.registration_status = 'ACTIVE'
      AND role.fantasy_role IS NOT NULL
)
SELECT
    COUNT(*) AS historical_identities,
    COUNT(*) FILTER (
        WHERE cc.player_id IS NULL
    ) AS historical_not_current_catalog,
    COUNT(*) FILTER (
        WHERE cc.player_id IS NOT NULL
    ) AS historical_still_current
FROM historical h
LEFT JOIN current_catalog cc
    ON cc.player_id = h.player_id;

    FROM players p
    JOIN player_reconciliation_results rec
        ON rec.player_id = p.id
    JOIN seasons s
        ON s.id = rec.season_id
       AND s.current_season = TRUE
    JOIN player_registrations registration
        ON registration.player_id = p.id
       AND registration.season_id = s.id
    JOIN season_clubs sc
        ON sc.season_id = s.id
       AND sc.club_id = registration.club_id
    JOIN player_roles role
        ON role.player_id = p.id
       AND role.season_id = s.id

    p.display_name,
    p.active,
    STRING_AGG(
        DISTINCT h.season,
        ', '

    ON cc.player_id = p.id

    p.display_name,
    p.active

WITH candidates AS (
    SELECT DISTINCT
        s.id AS season_id,
        s.code AS season,
        p.id AS player_id
    FROM seasons s
    JOIN player_roles role
        ON role.season_id = s.id
       AND role.source = 'FANTACALCIO'
    JOIN players p
        ON p.id = role.player_id
    WHERE s.current_season = FALSE

    UNION

    SELECT DISTINCT
        s.id,
        s.code,
        p.id
    FROM seasons s
    JOIN player_season_statistics stats
        ON stats.season_id = s.id
    JOIN players p
        ON p.id = stats.player_id
    WHERE s.current_season = FALSE
)
SELECT
    season,
    COUNT(DISTINCT player_id) AS catalog_candidates
FROM candidates
GROUP BY season_id, season
ORDER BY season;

WITH candidates AS (
    SELECT DISTINCT
        s.id AS season_id,
        s.code AS season,
        player_id
    FROM seasons s
    JOIN (
        SELECT season_id, player_id FROM player_roles
        WHERE source = 'FANTACALCIO'

        UNION

        SELECT season_id, player_id
        FROM player_season_statistics
    ) x
        ON x.season_id = s.id
    WHERE s.current_season = FALSE
)
SELECT
    c.season,
    COUNT(*) AS candidates,
    COUNT(*) FILTER (
        WHERE role.player_id IS NULL
    ) AS missing_fantacalcio_role,
    COUNT(*) FILTER (
        WHERE stats.player_id IS NULL
    ) AS missing_statistics
FROM candidates c
LEFT JOIN player_roles role
    ON role.player_id = c.player_id
   AND role.season_id = c.season_id
   AND role.source = 'FANTACALCIO'
LEFT JOIN (
    SELECT DISTINCT player_id, season_id
    FROM player_season_statistics
) stats
    ON stats.player_id = c.player_id
   AND stats.season_id = c.season_id
GROUP BY c.season
ORDER BY c.season;
