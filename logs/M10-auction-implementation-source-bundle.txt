function positiveInteger(value: string): number | null {
  const parsed = Number(value);
  return Number.isInteger(parsed) && parsed > 0 ? parsed : null;
}

function errorMessage(error: unknown): string | null {
  return error instanceof Error ? error.message : error ? 'Operazione non riuscita.' : null;
}

function newIdempotencyKey(sessionId: string): string {
  const random =
    typeof crypto.randomUUID === 'function'
      ? crypto.randomUUID()
      : `${Date.now()}-${Math.random().toString(36).slice(2)}`;
  return `live-auction-${sessionId}-${random}`;
}

export function LiveAuctionPage() {

    mutationFn: ({ playerId, price }: { playerId: number; price: number }) =>

    onError: (error) => setOperationError(error),

    mutationFn: (bid: number) => featureApi.sessionDecision(auth.getAuthorization, sessionId!, bid),
    onMutate: () => setOperationError(undefined),

      setDecision(undefined);

        buyerType === 'USER'

    onMutate: () => setOperationError(undefined),

    onSuccess: () => {

  return (

            </p>
          </div>

        {!sessionId ? (
          <section className={styles.panel}>
            <h2>Avvia o recupera una sessione</h2>
            <p>La sessione Ã¨ associata alla rosa e accessibile solo a owner o admin.</p>
            {squad.isPending ? (
              <p className={styles.muted}>Caricamento rosa personaleâ€¦</p>

                  Configura la rosa

            ) : squad.data ? (
              <div className={styles.grid}>

                  <span>Rosa</span>
                  <strong>{squad.data.name}</strong>

                <div className={styles.metric}>
                  <span>Giocatori</span>
                  <strong>
                    {squad.data.totalPlayers}/{squad.data.maximumPlayers}
                  </strong>

                <div className={styles.metric}>
                  <span>Budget residuo</span>
                  <strong>{squad.data.remainingBudget}</strong>
                </div>
              </div>
            ) : null}
            {squad.data && squad.data.totalPlayers === 0 && (
              <p className={styles.muted}>
                La rosa Ã¨ vuota: la sessione partirÃ  senza acquisti e userÃ  esclusivamente il

                <p className={styles.muted}>
                  Rosa in costruzione: mancano {squad.data.maximumPlayers - squad.data.totalPlayers}{' '}
                  giocatori.

            <div className={styles.actions}>

function params(values: Record<string, string | number | boolean | undefined>) {
  const p = new URLSearchParams();
  Object.entries(values).forEach(([key, value]) => {
    if (value !== undefined && value !== '') p.set(key, String(value));
  });
  return p.toString();
}
async function parsed<T>(
  path: string,
  schema: z.ZodType<T>,
  signal?: AbortSignal,
  getAuthorization?: () => string | undefined,
  init?: RequestInit,
) {
  return schema.parse(await httpRequest<unknown>(path, { ...init, signal, getAuthorization }));
}

@Service
public class AuctionApiPriceService {

    private final B67AuctionPriceModelService modelService;

    public AuctionApiPriceService(
            B67AuctionPriceModelService modelService
    ) {
        this.modelService = modelService;
    }

    /**
     * Public Auction API adapter.
     *
     * B67 is the current read model.
     * B51 remains immutable historical evidence.
     *
     * Public GET endpoints remain read-only and have no
     * persistence side effects.
     */
    @Transactional(readOnly = true)
    public AuctionPriceResponse calculate(
            Long playerId,
            String seasonCode,
            int budget,
            int teams,
            boolean defenseModifier
    ) {

        B67AuctionPriceModelService.AuctionPriceModelResult result =
                modelService.calculateAll(
                                seasonCode,
                                budget,
                                teams,
                                defenseModifier
                        )
                        .stream()
                        .filter(item ->
                                item.playerId().equals(
                                        playerId
                                )
                        )
                        .findFirst()
                        .orElseThrow(() ->
                                new ResourceNotFoundException(
                                        "Giocatore non trovato"
                                )
                        );

        if (!"PRICED".equals(result.pricingStatus())) {
            throw new IllegalArgumentException(
                    "Prezzo asta non disponibile: evidenza insufficiente"
            );
        }

        return toResponse(
                result,
                teams,
                budget,
                defenseModifier
        );
    }

    /**
     * Calculates the complete current B67 auction market once.
     * Used by ranking and downstream intelligence.
     */
    @Transactional(readOnly = true)
    public List<AuctionPriceResponse> calculateAll(
            String seasonCode,
            int budget,
            int teams,
            boolean defenseModifier
    ) {

        return modelService.calculateAll(
                        seasonCode,
                        budget,
                        teams,
                        defenseModifier
                )
                .stream()
                .filter(result ->
                        "PRICED".equals(
                                result.pricingStatus()
                        )
                )
                .map(result ->
                        toResponse(
                                result,
                                teams,
                                budget,
                                defenseModifier
                        )
                )
                .toList();
    }

    private AuctionPriceResponse toResponse(
            B67AuctionPriceModelService.AuctionPriceModelResult result,
            int teams,
            int budget,
            boolean defenseModifier
    ) {

        BigDecimal calculationScore =
                BigDecimal
                        .valueOf(
                                result.auctionScore()
                        )
                        .setScale(
                                2,
                                RoundingMode.HALF_UP
                        );

        String explanation =
                String.format(
                        "Prezzo calcolato con %s. "
                                + "Score asta %.2f/100, "
                                + "rank ruolo %d, "
                                + "percentile ruolo %.2f/100, "
                                + "evidenza %s.",
                        B67AuctionPriceModelService.ALGORITHM_VERSION,
                        result.auctionScore(),
                        result.auctionRoleRank(),
                        result.auctionRolePercentile(),
                        result.evidenceTier()
                );

        return new AuctionPriceResponse(
                result.playerId(),
                result.playerName(),
                result.club(),
                result.role(),
                teams,
                budget,
                defenseModifier,
                result.minimumPrice(),
                result.recommendedPrice(),
                result.maximumPrice(),
                result.budgetPercentage(),
                calculationScore,
                result.pricingStatus(),
                explanation
        );
    }
}

package it.fantacalciointelligence.backend.auction;

import java.util.Comparator;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class AuctionRankingService {

    private final JdbcClient jdbcClient;
    private final AuctionApiPriceService auctionApiPriceService;

    public AuctionRankingService(
            JdbcClient jdbcClient,
            AuctionApiPriceService auctionApiPriceService
    ) {
        this.jdbcClient = jdbcClient;
        this.auctionApiPriceService = auctionApiPriceService;
    }

    @Transactional(readOnly = true)
    public List<AuctionRankingResponse> calculateRanking(
            String season,
            int budget,
            int teams,
            boolean defenseModifier,
            String role,
            String club,
            int limit
    ) {
        int safeLimit = Math.min(
                Math.max(limit, 1),
                500
        );

        StringBuilder sql = new StringBuilder("""
                SELECT DISTINCT
                    p.id
                FROM players p

                JOIN player_reconciliation_results reconciliation
                    ON reconciliation.player_id = p.id

                JOIN seasons s
                    ON s.id = reconciliation.season_id

                JOIN player_registrations registration
                    ON registration.player_id = p.id
                    AND registration.season_id = s.id

                JOIN clubs c
                    ON c.id = registration.club_id

                JOIN season_clubs season_club
                    ON season_club.season_id = s.id
                    AND season_club.club_id = c.id

                JOIN player_roles player_role
                    ON player_role.player_id = p.id
                    AND player_role.season_id = s.id
                    AND player_role.source = 'FANTACALCIO'

                WHERE s.code = :season
                  AND p.active = TRUE
                  AND registration.registration_status = 'ACTIVE'

                """);

        if (role != null && !role.isBlank()) {
            sql.append("""
                    AND player_role.fantasy_role = :role
                    """);
        }

        if (club != null && !club.isBlank()) {
            sql.append("""
                    AND (
                        LOWER(c.slug) = LOWER(:club)
                        OR LOWER(c.name) = LOWER(:club)
                    )
                    """);
        }

        sql.append("""
                ORDER BY p.id
                """);

        var query = jdbcClient
                .sql(sql.toString())
                .param("season", season);

        if (role != null && !role.isBlank()) {
            query = query.param(
                    "role",
                    role.trim().toUpperCase()
            );
        }

        if (club != null && !club.isBlank()) {
            query = query.param(
                    "club",
                    club.trim()
            );
        }

        Set<Long> allowedPlayerIds =
                new HashSet<>(
                        query
                                .query(Long.class)
                                .list()
                );

        if (allowedPlayerIds.isEmpty()) {
            return List.of();
        }

        List<AuctionPriceResponse> prices =
                auctionApiPriceService.calculateAll(
                                season,
                                budget,
                                teams,
                                defenseModifier
                        )
                        .stream()
                        .filter(price ->
                                allowedPlayerIds.contains(
                                        price.playerId()
                                )
                        )
                        .sorted(
                                Comparator
                                        .comparing(
                                                AuctionPriceResponse
                                                        ::calculationScore
                                        )
                                        .reversed()
                                        .thenComparing(
                                                AuctionPriceResponse
                                                        ::recommendedPrice,
                                                Comparator.reverseOrder()
                                        )
                                        .thenComparing(
                                                AuctionPriceResponse
                                                        ::playerName
                                        )
                        )
                        .limit(safeLimit)
                        .toList();

        return java.util.stream.IntStream
                .range(
                        0,
                        prices.size()
                )
                .mapToObj(index -> {
                    AuctionPriceResponse price =
                            prices.get(index);

                    return new AuctionRankingResponse(
                            index + 1,
                            price.playerId(),
                            price.playerName(),
                            price.club(),
                            price.role(),
                            price.minimumPrice(),
                            price.recommendedPrice(),
                            price.maximumPrice(),
                            price.budgetPercentage(),
                            price.calculationScore(),
                            price.recommendation()
                    );
                })
                .toList();
    }
}

package it.fantacalciointelligence.backend.auction;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.function.Function;

import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Service;

@Service
public class B67AuctionPriceModelService {

    public static final String ALGORITHM_VERSION =
            "B67_AUCTION_PRICE_V1";

    public static final String PERFORMANCE_VERSION =
            "B66_FEATURES_V3";

    public static final String MODEL_FAMILY =
            "B66_MARKET_ROLE_RELATIVE_AUCTION_PRICE";

    private final JdbcClient jdbcClient;
    private final B67AuctionValueScorer valueScorer;

    public B67AuctionPriceModelService(
            JdbcClient jdbcClient,
            B67AuctionValueScorer valueScorer
    ) {
        this.jdbcClient = jdbcClient;
        this.valueScorer = valueScorer;
    }

    public List<AuctionPriceModelResult> calculateAll(
            String seasonCode,
            int budget,
            int teams,
            boolean defenseModifier
    ) {
        validateConfiguration(
                budget,
                teams
        );

        List<PlayerInput> players =
                loadPlayers(seasonCode);

        if (players.isEmpty()) {
            throw new IllegalArgumentException(
                    "Nessun giocatore operativo B67 disponibile per la stagione "
                            + seasonCode
            );
        }

        Map<String, List<PlayerInput>> byRole =
                groupByRole(players);

        List<AuctionPriceModelResult> results =
                new ArrayList<>();

        for (String role : List.of("P", "D", "C", "A")) {

            List<PlayerInput> rolePlayers =
                    byRole.getOrDefault(
                            role,
                            List.of()
                    );

            if (rolePlayers.isEmpty()) {
                continue;
            }

            results.addAll(
                    calculateRole(
                            role,
                            rolePlayers,
                            budget,
                            teams,
                            defenseModifier
                    )
            );
        }

        return results;
    }

    private List<AuctionPriceModelResult> calculateRole(
            String role,
            List<PlayerInput> players,
            int budget,
            int teams,
            boolean defenseModifier
    ) {

        Map<Long, Double> fvmPercentiles =
                calculatePercentiles(
                        players,
                        PlayerInput::fantasyMarketValue
                );

        Map<Long, Double> quotationPercentiles =
                calculatePercentiles(
                        players,
                        PlayerInput::quotation
                );

        List<ScoredPlayer> scored =
                new ArrayList<>();

        for (PlayerInput player : players) {

            Double fvmPercentile =
                    fvmPercentiles.get(
                            player.playerId()
                    );

            Double quotationPercentile =
                    quotationPercentiles.get(
                            player.playerId()
                    );

            B67AuctionValueScorer.ScoreResult scoreResult =
                    valueScorer.score(
                            role,
                            fvmPercentile,
                            quotationPercentile,
                            player.fantasyAverageRolePercentile(),
                            player.usageRolePercentile(),
                            player.goalRateRolePercentile(),
                            player.assistRateRolePercentile()
                    );

            if (scoreResult.auctionValueScore() == null) {
                throw new IllegalStateException(
                        "B67 senza evidenza reale per player_id="
                                + player.playerId()
                );
            }

            Double marketSignal =
                    valueScorer.marketSignal(
                            fvmPercentile,
                            quotationPercentile
                    );

            String evidenceTier =
                    evidenceTier(
                            player,
                            scoreResult.performanceSignalCount()
                    );

            scored.add(
                    new ScoredPlayer(
                            player,
                            round2(fvmPercentile),
                            round2(quotationPercentile),
                            round2(marketSignal),
                            round2(scoreResult.auctionValueScore()),
                            scoreResult.performanceSignalCount(),
                            evidenceTier
                    )
            );
        }

        scored.sort(
                Comparator
                        .comparing(
                                ScoredPlayer::auctionScore
                        )
                        .reversed()
                        .thenComparing(
                                item ->
                                        item.player().playerId()
                        )
        );

        int marketSlots =
                teams * roleSlots(role);

        List<ScoredPlayer> contenders =
                scored.stream()
                        .limit(marketSlots)
                        .toList();

        double leagueRoleBudget =
                budget
                        * teams
                        * roleAllocation(
                                role,
                                defenseModifier
                        );

        double budgetFactor =
                budget / 1000.0;

        double teamFactor =
                1.0
                        + (
                                (teams - 8)
                                        * 0.05
                        );

        Map<Long, Double> provisionalPrices =
                new HashMap<>();

        for (ScoredPlayer player : contenders) {

            double rawPrice =
                    interpolatePrice(
                            role,
                            player.auctionScore()
                    );

            rawPrice *= budgetFactor;
            rawPrice *= teamFactor;

            provisionalPrices.put(
                    player.player().playerId(),
                    Math.max(
                            1.0,
                            rawPrice
                    )
            );
        }

        double provisionalTotal =
                provisionalPrices
                        .values()
                        .stream()
                        .mapToDouble(Double::doubleValue)
                        .sum();

        boolean budgetCapApplied =
                provisionalTotal > leagueRoleBudget;

        if (budgetCapApplied) {

            int floors =
                    provisionalPrices.size();

            double distributableBudget =
                    Math.max(
                            0,
                            leagueRoleBudget - floors
                    );

            double rawExtra =
                    provisionalPrices
                            .values()
                            .stream()
                            .mapToDouble(
                                    price ->
                                            Math.max(
                                                    0,
                                                    price - 1
                                            )
                            )
                            .sum();

            double scale =
                    rawExtra > 0
                            ? distributableBudget / rawExtra
                            : 0;

            provisionalPrices.replaceAll(
                    (playerId, price) ->
                            1
                                    + Math.max(
                                            0,
                                            price - 1
                                    )
                                    * scale
            );
        }

        Map<Long, Integer> recommendedPrices =
                new HashMap<>();

        /*
         * Every operational Fantacalcio player receives
         * an actionable market price.
         *
         * Players outside the active market slots are priced at 1.
         */
        for (ScoredPlayer scoredPlayer : scored) {
            recommendedPrices.put(
                    scoredPlayer.player().playerId(),
                    1
            );
        }

        if (budgetCapApplied) {

            int integerBudgetTarget =
                    (int) Math.floor(
                            leagueRoleBudget
                                    + 0.000000001
                    );

            List<PriceRemainder> remainders =
                    new ArrayList<>();

            int assignedCredits = 0;

            for (ScoredPlayer contender : contenders) {

                Long playerId =
                        contender.player().playerId();

                double exactPrice =
                        provisionalPrices.get(
                                playerId
                        );

                int floorPrice =
                        Math.max(
                                1,
                                (int) Math.floor(
                                        exactPrice
                                )
                        );

                recommendedPrices.put(
                        playerId,
                        floorPrice
                );

                assignedCredits += floorPrice;

                remainders.add(
                        new PriceRemainder(
                                playerId,
                                exactPrice - floorPrice,
                                contender.auctionScore()
                        )
                );
            }

            int creditsToDistribute =
                    Math.max(
                            0,
                            integerBudgetTarget
                                    - assignedCredits
                    );

            remainders.sort(
                    Comparator
                            .comparing(
                                    PriceRemainder::fraction
                            )
                            .reversed()
                            .thenComparing(
                                    PriceRemainder::auctionScore,
                                    Comparator.reverseOrder()
                            )
                            .thenComparing(
                                    PriceRemainder::playerId
                            )
            );

            int distributableCredits =
                    Math.min(
                            creditsToDistribute,
                            remainders.size()
                    );

            for (
                    int index = 0;
                    index < distributableCredits;
                    index++
            ) {
                Long playerId =
                        remainders
                                .get(index)
                                .playerId();

                recommendedPrices.computeIfPresent(
                        playerId,
                        (ignored, currentPrice) ->
                                currentPrice + 1
                );
            }

        } else {

            for (ScoredPlayer contender : contenders) {

                Long playerId =
                        contender.player().playerId();

                int roundedPrice =
                        Math.max(
                                1,
                                (int) Math.round(
                                        provisionalPrices.get(
                                                playerId
                                        )
                                )
                        );

                recommendedPrices.put(
                        playerId,
                        roundedPrice
                );
            }
        }

        List<AuctionPriceModelResult> results =
                new ArrayList<>();

        int position = 0;
        int auctionRank = 0;
        Double previousAuctionScore = null;

        for (ScoredPlayer scoredPlayer : scored) {

            position++;

            if (
                    previousAuctionScore == null
                    || Double.compare(
                            scoredPlayer.auctionScore(),
                            previousAuctionScore
                    ) != 0
            ) {
                auctionRank = position;
            }

            previousAuctionScore =
                    scoredPlayer.auctionScore();

            PlayerInput player =
                    scoredPlayer.player();

            int recommendedPrice =
                    recommendedPrices.getOrDefault(
                            player.playerId(),
                            1
                    );

            PriceRange range =
                    calculateRange(
                            recommendedPrice,
                            player.dataConfidenceScore(),
                            player.fantasyMarketValue() != null
                    );

            double auctionPercentile =
                    competitionPercentile(
                            auctionRank,
                            scored.size()
                    );

            results.add(
                    new AuctionPriceModelResult(
                            player.playerId(),
                            player.playerName(),
                            player.club(),
                            player.role(),
                            round2(
                                    player.dataConfidenceScore()
                            ),
                            player.historySeasonsAvailable(),
                            player.sourceMix(),
                            player.quotation(),
                            player.fantasyMarketValue(),
                            scoredPlayer.fvmRolePercentile(),
                            scoredPlayer.quotationRolePercentile(),
                            round2(
                                    player.fantasyAverageRolePercentile()
                            ),
                            round2(
                                    player.usageRolePercentile()
                            ),
                            round2(
                                    player.goalRateRolePercentile()
                            ),
                            round2(
                                    player.assistRateRolePercentile()
                            ),
                            scoredPlayer.marketSignal(),
                            scoredPlayer.performanceSignalCount(),
                            scoredPlayer.evidenceTier(),
                            scoredPlayer.auctionScore(),
                            auctionRank,
                            round2(
                                    auctionPercentile
                            ),
                            "PRICED",
                            range.minimumPrice(),
                            recommendedPrice,
                            range.maximumPrice(),
                            BigDecimal
                                    .valueOf(
                                            recommendedPrice
                                                    * 100.0
                                                    / budget
                                    )
                                    .setScale(
                                            3,
                                            RoundingMode.HALF_UP
                                    )
                    )
            );
        }

        return results;
    }

    private Map<Long, Double> calculatePercentiles(
            List<PlayerInput> players,
            Function<PlayerInput, Double> extractor
    ) {

        List<Double> values =
                players.stream()
                        .map(extractor)
                        .filter(value -> value != null)
                        .sorted()
                        .toList();

        Map<Long, Double> percentiles =
                new HashMap<>();

        int n = values.size();

        if (n == 0) {
            return percentiles;
        }

        for (PlayerInput player : players) {

            Double value =
                    extractor.apply(player);

            if (value == null) {
                continue;
            }

            long lower =
                    values.stream()
                            .filter(item ->
                                    item < value
                            )
                            .count();

            long equal =
                    values.stream()
                            .filter(item ->
                                    Double.compare(
                                            item,
                                            value
                                    ) == 0
                            )
                            .count();

            double percentile;

            if (n == 1) {
                percentile = 100.0;

            } else {

                double averageRankZeroBased =
                        lower
                                + (
                                        equal - 1
                                ) / 2.0;

                percentile =
                        averageRankZeroBased
                                / (n - 1)
                                * 100.0;
            }

            percentiles.put(
                    player.playerId(),
                    percentile
            );
        }

        return percentiles;
    }

    private String evidenceTier(
            PlayerInput player,
            int performanceSignalCount
    ) {

        if (
                player.historySeasonsAvailable() != null
                && player.historySeasonsAvailable() == 0
        ) {
            return "MARKET_ONLY";
        }

        double confidence =
                player.dataConfidenceScore() == null
                        ? 0
                        : player.dataConfidenceScore();

        if (
                confidence >= 80
                && performanceSignalCount >= 3
        ) {
            return "HIGH_EVIDENCE";
        }

        if (
                confidence >= 70
                && performanceSignalCount >= 1
        ) {
            return "MEDIUM_EVIDENCE";
        }

        return "LOW_EVIDENCE";
    }

    /*
     * Price curve inherited from certified B51.
     *
     * B67 changes the upstream value evidence, not the
     * already-tested conversion between role-relative value
     * and auction credits.
     */
    private double interpolatePrice(
            String role,
            double percentile
    ) {

        double[][] anchors =
                priceAnchors(role);

        if (percentile >= anchors[0][0]) {
            return anchors[0][1];
        }

        if (
                percentile
                        <= anchors[
                                anchors.length - 1
                        ][0]
        ) {
            return anchors[
                    anchors.length - 1
            ][1];
        }

        for (
                int index = 0;
                index < anchors.length - 1;
                index++
        ) {

            double highPercentile =
                    anchors[index][0];

            double highPrice =
                    anchors[index][1];

            double lowPercentile =
                    anchors[index + 1][0];

            double lowPrice =
                    anchors[index + 1][1];

            if (
                    percentile <= highPercentile
                    && percentile >= lowPercentile
            ) {

                double span =
                        highPercentile
                                - lowPercentile;

                if (span == 0) {
                    return highPrice;
                }

                double position =
                        (
                                percentile
                                        - lowPercentile
                        ) / span;

                return lowPrice
                        + position
                        * (
                                highPrice
                                        - lowPrice
                        );
            }
        }

        return 1.0;
    }

    private PriceRange calculateRange(
            double price,
            double confidence,
            boolean hasFvm
    ) {

        double lowFactor;
        double highFactor;

        if (confidence >= 90) {
            lowFactor = 0.90;
            highFactor = 1.10;

        } else if (confidence >= 80) {
            lowFactor = 0.85;
            highFactor = 1.15;

        } else if (confidence >= 70) {
            lowFactor = 0.80;
            highFactor = 1.20;

        } else {
            lowFactor = 0.72;
            highFactor = 1.28;
        }

        if (!hasFvm) {
            lowFactor -= 0.05;
            highFactor += 0.05;
        }

        int minimum =
                Math.max(
                        1,
                        (int) Math.round(
                                price * lowFactor
                        )
                );

        int maximum =
                Math.max(
                        (int) Math.round(price),
                        (int) Math.round(
                                price * highFactor
                        )
                );

        return new PriceRange(
                minimum,
                maximum
        );
    }

    private double competitionPercentile(
            int rank,
            int total
    ) {

        if (total <= 1) {
            return 100.0;
        }

        return (
                (total - rank)
                        * 100.0
                        / (total - 1)
        );
    }

        }
    }

    @Transactional
    public AuctionRankingResponse findRanking(
            String season,
            int budget,
            int teams,
            boolean defenseModifier,
            String role,
            String club,
            Integer maximumPrice,
            String recommendation,
            boolean excludeUnavailable,
            int limit
    ) {
        RankingRequest request =
                validateAndNormalize(
                        season,
                        budget,
                        teams,
                        role,
                        club,
                        maximumPrice,
                        recommendation,
                        limit
                );

        List<Long> playerIds =
                findEligiblePlayerIds(
                        request.season(),
                        request.role(),
                        request.club()
                );

        List<PlayerIntelligenceResponse> profiles =
                new ArrayList<>();

        if (playerIds.isEmpty()) {
            return emptyResponse(
                    request,
                    budget,
                    teams,
                    defenseModifier,
                    excludeUnavailable
            );
        }

        Map<Long, AuctionPriceResponse> pricesByPlayer =
                auctionPriceService.calculateAll(
                                request.season(),
                                budget,
                                teams,

                        )
                        .stream()
                        .collect(Collectors.toMap(
                                AuctionPriceResponse::playerId,
                                Function.identity()
                        ));

            AuctionPriceResponse price =
                    pricesByPlayer.get(playerId);

            // An absent entry is explicitly INSUFFICIENT_EVIDENCE in B51.
            if (price == null) {
                continue;
            }

            PlayerIntelligenceResponse profile =
                    intelligenceService.findByPlayer(
                            playerId,
                            request.season(),
                            budget,
                            teams,
                            defenseModifier,
                            price
                    );

            if (matchesFilters(
                    profile,
                    request.maximumPrice(),
                    request.recommendation(),
                    excludeUnavailable
            )) {
                profiles.add(profile);
            }
        }

        profiles.sort(
                Comparator
                        .comparing(
                                PlayerIntelligenceResponse::auctionScore,
                                Comparator.nullsLast(
                                        Comparator.reverseOrder()
                                )
                        )
                        .thenComparing(
                                PlayerIntelligenceResponse
                                        ::recommendedPrice,
                                Comparator.reverseOrder()
                        )
                        .thenComparing(
                                PlayerIntelligenceResponse::playerName,
                                String.CASE_INSENSITIVE_ORDER
                        )
        );

        List<PlayerIntelligenceResponse> limitedProfiles =
                profiles.stream()
                        .limit(request.limit())
                        .toList();

        List<AuctionRankingItemResponse> items =
                new ArrayList<>();

        for (int index = 0;
                index < limitedProfiles.size();
                index++) {

            PlayerIntelligenceResponse profile =
                    limitedProfiles.get(index);

            items.add(
                    new AuctionRankingItemResponse(
                            index + 1,
                            profile.playerId(),
                            profile.playerName(),
                            profile.club(),
                            profile.fantasyRole(),
                            safeDecimal(profile.auctionScore()),
                            profile.minimumPrice(),
                            profile.recommendedPrice(),
                            profile.maximumPrice(),
                            safeDecimal(
                                    profile.budgetPercentage()
                            ),
                            safeDecimal(
                                    profile.reliabilityScore()
                            ),
                            safeDecimal(
                                    profile.potentialScore()
                            ),
                            safeDecimal(
                                    profile.riskScore()
                            ),
                            safeDecimal(
                                    profile.starterProbability()
                            ),
                            profile.availabilityStatus(),
                            profile.availabilityMatchday(),
                            profile.strengths(),
                            profile.risks(),
                            profile.finalRecommendation()
                    )
            );
        }

        return new AuctionRankingResponse(
                request.season(),
                budget,
                teams,
                defenseModifier,
                request.role(),
                request.club(),
                request.maximumPrice(),
                request.recommendation(),

    private List<Long> findEligiblePlayerIds(
            String season,
            String role,
            String club
    ) {
        StringBuilder sql = new StringBuilder("""
                SELECT DISTINCT player.id

                FROM players player

                JOIN player_registrations registration
                    ON registration.player_id = player.id
                    AND registration.registration_status =
                        'ACTIVE'

                JOIN seasons season
                    ON season.id = registration.season_id

                JOIN clubs club
                    ON club.id = registration.club_id

                JOIN season_clubs season_club
                    ON season_club.season_id = season.id
                    AND season_club.club_id = club.id

                JOIN player_roles role
                    ON role.player_id = player.id
                    AND role.season_id = season.id
                    AND role.source = 'FANTACALCIO'


                WHERE player.active = TRUE
                  AND season.code = :season
                """);

        if (role != null) {
            sql.append("""
                    AND role.fantasy_role = :role
                    """);
        }

        if (club != null) {
            sql.append("""
                    AND (
                        LOWER(club.name) = :club
                        OR LOWER(club.slug) = :club
                    )
                    """);
        }

        sql.append("""
                ORDER BY player.id
                """);

        var query =
                jdbcClient.sql(sql.toString())
                        .param("season", season);

        if (role != null) {
            query = query.param("role", role);
        }

        if (club != null) {
            query = query.param("club", club);
        }

        return query
                .query(Long.class)
                .list();
    }

    private boolean matchesFilters(
            PlayerIntelligenceResponse profile,
            Integer maximumPrice,
            String recommendation,
            boolean excludeUnavailable
    ) {
        if (maximumPrice != null
                && profile.recommendedPrice() > maximumPrice) {
            return false;
        }

        if (recommendation != null
                && !recommendation.equals(
                        profile.finalRecommendation()
                )) {
            return false;
        }

        if (excludeUnavailable
                && profile.availabilityStatus() != null
                && UNAVAILABLE_STATUSES.contains(
                        profile.availabilityStatus()
                                .toUpperCase(Locale.ROOT)
                )) {
            return false;
        }

        return true;
    }

    private RankingRequest validateAndNormalize(
            String season,
            int budget,
            int teams,
            String role,
            String club,
            Integer maximumPrice,
            String recommendation,
            int limit
    ) {
        if (season == null || season.isBlank()) {
            throw new BadRequestException(
                    "La stagione Ã¨ obbligatoria"
            );
        }

        if (budget < 150 || budget > 1000) {
            throw new BadRequestException(
                    "Il budget deve essere compreso tra 150 e 1000"
            );
        }

        if (teams < 6 || teams > 12) {
            throw new BadRequestException(
                    "Le squadre devono essere comprese tra 6 e 12"
            );
        }

        String normalizedRole =
                normalizeNullable(role);

        if (normalizedRole != null) {
            normalizedRole =
                    normalizedRole.toUpperCase(Locale.ROOT);

            if (!ROLES.contains(normalizedRole)) {
                throw new BadRequestException(
                        "Il ruolo deve essere P, D, C oppure A"
                );
            }
        }

        String normalizedClub =
                normalizeNullable(club);

        if (normalizedClub != null) {
            normalizedClub =
                    normalizedClub.toLowerCase(Locale.ROOT);
        }

        if (maximumPrice != null
                && maximumPrice < 1) {
            throw new BadRequestException(
                    "Il prezzo massimo deve essere positivo"
            );
        }

        if (maximumPrice != null
                && maximumPrice > budget) {
            throw new BadRequestException(
                    "Il prezzo massimo non puÃ² superare il budget"
            );
        }

        String normalizedRecommendation =
                normalizeNullable(recommendation);

        if (normalizedRecommendation != null) {
            normalizedRecommendation =
                    normalizedRecommendation
                            .toUpperCase(Locale.ROOT);

            if (!RECOMMENDATIONS.contains(
                    normalizedRecommendation
            )) {
                throw new BadRequestException(
                        "Raccomandazione non valida"
                );
            }
        }

        if (limit < 1 || limit > 100) {
            throw new BadRequestException(
                    "Il limite deve essere compreso tra 1 e 100"
            );
        }

        return new RankingRequest(
                season.trim(),
                normalizedRole,
                normalizedClub,
                maximumPrice,
                normalizedRecommendation,
                limit
        );
    }

    private String normalizeNullable(String value) {
        if (value == null || value.isBlank()) {
            return null;
        }

        return value.trim();
    }

    private BigDecimal safeDecimal(
            BigDecimal value
    ) {
        return value == null
                ? BigDecimal.ZERO
                : value;
    }

            String season,
            String role,
            String club,

    ) {
    }

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import it.fantacalciointelligence.backend.common.BadRequestException;

@Service
public class IntelligentAuctionStrategyService {

    private static final Map<String, Integer> REQUIRED_SLOTS =
            Map.of(
                    "P", 3,
                    "D", 8,
                    "C", 8,
                    "A", 6
            );

    private final IntelligentAuctionRankingService rankingService;

    public IntelligentAuctionStrategyService(
            IntelligentAuctionRankingService rankingService

        this.rankingService = rankingService;

    @Transactional
    public AuctionStrategyResponse buildStrategy(
            String season,
            int budget,
            int teams,
            boolean defenseModifier
    ) {
        validateRequest(season, budget, teams);

        String normalizedSeason = season.trim();

        Map<String, BigDecimal> allocations =
                roleAllocations(defenseModifier);

        List<AuctionRoleStrategyResponse> roleStrategies =
                new ArrayList<>();

        List<String> globalWarnings =
                new ArrayList<>();

        int totalPlannedSpending = 0;

        for (String role : List.of("P", "D", "C", "A")) {
            AuctionRoleStrategyResponse roleStrategy =
                    buildRoleStrategy(
                            normalizedSeason,
                            budget,
                            teams,
                            defenseModifier,
                            role,
                            allocations.get(role)
                    );

            roleStrategies.add(roleStrategy);

            totalPlannedSpending +=
                    roleStrategy.plannedSpending();

            if (roleStrategy.primaryTargets().isEmpty()) {
                globalWarnings.add(
                        "Nessun obiettivo principale disponibile "
                                + "per il ruolo "
                                + role
                );
            }
        }

        int remainingBudget =
                budget - totalPlannedSpending;

        boolean withinBudget =
                remainingBudget >= 0;

        if (!withinBudget) {
            globalWarnings.add(
                    "Il piano supera il budget disponibile di "
                            + Math.abs(remainingBudget)
                            + " crediti"
            );
        }

        if (remainingBudget > budget * 0.20) {
            globalWarnings.add(
                    "Il piano conserva una quota elevata di budget: "
                            + "valutare un investimento maggiore "
                            + "sugli obiettivi principali"
            );
        }

        String summary =
                buildSummary(
                        budget,
                        totalPlannedSpending,
                        remainingBudget,
                        defenseModifier,
                        withinBudget
                );

        return new AuctionStrategyResponse(
                normalizedSeason,
                budget,
                teams,
                defenseModifier,
                25,
                totalPlannedSpending,
                remainingBudget,
                withinBudget,
                summary,
                List.copyOf(roleStrategies),
                List.copyOf(globalWarnings)
        );
    }

    private AuctionRoleStrategyResponse buildRoleStrategy(
            String season,
            int totalBudget,
            int teams,
            boolean defenseModifier,
            String role,
            BigDecimal allocation
    ) {
        int requiredSlots =
                REQUIRED_SLOTS.get(role);

        int allocatedBudget =
                allocation
                        .multiply(
                                BigDecimal.valueOf(totalBudget)
                        )
                        .setScale(
                                0,
                                RoundingMode.HALF_UP
                        )
                        .intValue();

        AuctionRankingResponse ranking =
                rankingService.findRanking(
                        season,
                        totalBudget,
                        teams,
                        defenseModifier,
                        role,
                        null,
                        null,
                        null,
                        true,
                        100
                );

        List<AuctionRankingItemResponse> players =
                ranking.players();

        List<AuctionStrategyPlayerResponse> primaryTargets =
                players.stream()
                        .filter(this::isPrimaryTarget)
                        .limit(primaryTargetCount(role))
                        .map(player ->
                                toStrategyPlayer(
                                        player,
                                        "OBIETTIVO PRINCIPALE"
                                )
                        )
                        .toList();

        List<AuctionStrategyPlayerResponse> alternatives =
                players.stream()
                        .filter(player ->
                                isValueAlternative(
                                        player,
                                        allocatedBudget,
                                        requiredSlots
                                )
                        )
                        .filter(player ->
                                !containsPlayer(
                                        primaryTargets,
                                        player.playerId()
                                )
                        )
                        .limit(alternativeCount(role))
                        .map(player ->
                                toStrategyPlayer(
                                        player,
                                        "ALTERNATIVA ECONOMICA"
                                )
                        )
                        .toList();

        List<AuctionStrategyPlayerResponse> bets =
                players.stream()
                        .filter(this::isHighPotentialBet)
                        .filter(player ->
                                !containsPlayer(
                                        primaryTargets,
                                        player.playerId()
                                )
                        )
                        .filter(player ->
                                !containsPlayer(
                                        alternatives,
                                        player.playerId()
                                )
                        )
                        .limit(2)
                        .map(player ->
                                toStrategyPlayer(
                                        player,
                                        "SCOMMESSA"
                                )
                        )
                        .toList();

        List<AuctionStrategyPlayerResponse> avoid =
                players.stream()
                        .filter(this::shouldAvoid)
                        .limit(3)
                        .map(player ->
                                toStrategyPlayer(
                                        player,
                                        "DA EVITARE"
                                )
                        )
                        .toList();

        int plannedSpending =
                calculatePlannedSpending(
                        allocatedBudget,
                        requiredSlots,
                        primaryTargets,
                        alternatives,
                        bets
                );

        int remainingRoleBudget =
                allocatedBudget - plannedSpending;

        List<String> warnings =
                buildRoleWarnings(
                        role,
                        requiredSlots,
                        allocatedBudget,
                        plannedSpending,
                        players,
                        primaryTargets
                );

        return new AuctionRoleStrategyResponse(
                role,
                requiredSlots,
                allocation
                        .multiply(BigDecimal.valueOf(100))
                        .setScale(1, RoundingMode.HALF_UP),
                allocatedBudget,
                plannedSpending,
                remainingRoleBudget,
                primaryTargets,
                alternatives,
                bets,
                avoid,
                warnings
        );
    }

    private int calculatePlannedSpending(
            int allocatedBudget,
            int requiredSlots,
            List<AuctionStrategyPlayerResponse> primaryTargets,
            List<AuctionStrategyPlayerResponse> alternatives,
            List<AuctionStrategyPlayerResponse> bets
    ) {
        List<AuctionStrategyPlayerResponse> selected =
                new ArrayList<>();

        selected.addAll(primaryTargets);
        selected.addAll(alternatives);
        selected.addAll(bets);

        int spending =
                selected.stream()
                        .limit(requiredSlots)
                        .mapToInt(
                                AuctionStrategyPlayerResponse
                                        ::recommendedPrice
                        )
                        .sum();

        int selectedSlots =
                Math.min(
                        selected.size(),
                        requiredSlots
                );

        int minimumReserve =
                Math.max(
                        0,
                        requiredSlots - selectedSlots
                );

        spending += minimumReserve;

        return Math.min(
                spending,
                allocatedBudget
        );
    }

    private boolean isPrimaryTarget(
            AuctionRankingItemResponse player
    ) {
        return (
                "OBIETTIVO PRIORITARIO".equals(
                        player.finalRecommendation()
                )
                || "ACQUISTO CONSIGLIATO".equals(
                        player.finalRecommendation()
                )
        )
                && decimal(player.riskScore()) <= 60
                && decimal(player.starterProbability()) >= 60;
    }

    private boolean isValueAlternative(
            AuctionRankingItemResponse player,
            int allocatedBudget,
            int requiredSlots
    ) {
        double averageSlotBudget =
                allocatedBudget / (double) requiredSlots;

        return player.recommendedPrice()
                <= Math.max(
                        1,
                        Math.round(averageSlotBudget)
                )
                && decimal(player.riskScore()) < 70
                && !"ATTENDERE AGGIORNAMENTI".equals(
                        player.finalRecommendation()
                );
    }

    private boolean isHighPotentialBet(
            AuctionRankingItemResponse player
    ) {
        return decimal(player.potentialScore()) >= 70
                && player.recommendedPrice()
                        <= player.maximumPrice()
                && decimal(player.riskScore()) < 80
                && !"ATTENDERE AGGIORNAMENTI".equals(
                        player.finalRecommendation()
                );
    }

    private boolean shouldAvoid(
            AuctionRankingItemResponse player
    ) {
        return decimal(player.riskScore()) >= 70
                || "EVITARE O PRENDERE A BASSO COSTO".equals(
                        player.finalRecommendation()
                )
                || "ATTENDERE AGGIORNAMENTI".equals(
                        player.finalRecommendation()
                );
    }

    }

    private List<String> buildRoleWarnings(
            String role,
            int requiredSlots,
            int allocatedBudget,
            int plannedSpending,
            List<AuctionRankingItemResponse> availablePlayers,
            List<AuctionStrategyPlayerResponse> primaryTargets
    ) {

                    "Disponibili soltanto "
                            + availablePlayers.size()
                            + " giocatori per "
                            + requiredSlots
                            + " slot richiesti"
            );
        }

        if (primaryTargets.isEmpty()) {
            warnings.add(
                    "Nessun profilo di prioritÃ  alta individuato"
            );
        }

        if (plannedSpending > allocatedBudget) {
            warnings.add(
                    "La spesa prevista supera il budget del ruolo"
            );
        }

        if ("D".equals(role)

            warnings.add(
                    "Mantenere almeno due difensori affidabili "
                            + "per proteggere il modificatore"
            );
        }

    }

    private boolean containsPlayer(
            List<AuctionStrategyPlayerResponse> players,

    ) {
        return players.stream()

                        player.playerId().equals(playerId)

    private int primaryTargetCount(String role) {

    }

                );
            }
            else if (remaining == 1) {
                coverageStatus =
                        "QUASI COMPLETO";
            }
            else {
                coverageStatus =
                        "IN COSTRUZIONE";
            }

            if ("D".equals(role)
                    && current < 4) {

                warnings.add(
                        "Copertura difensiva ancora insufficiente "
                                + "per gestire rotazioni e modificatore"
                );
            }

            if ("P".equals(role)
                    && current == 1) {

                warnings.add(
                        "Ãˆ presente un solo portiere: "
                                + "manca una riserva"
                );
            }

            response.add(
                    new SquadConstraintRoleResponse(
                            role,
                            current,
                            maximum,
                            remaining,
                            complete,
                            coverageStatus,
                            List.copyOf(warnings)
                    )
            );
        }

        return response;
    }

    private List<SquadCalendarPlayerResponse>
    findCalendarPlayers(
            SquadResponse squad,
            int fromMatchday,
            int toMatchday,
            List<String> globalWarnings
    ) {
        List<PlayerRow> rows =
                jdbcClient.sql("""
                        SELECT
                            squad_player.player_id,
                            player.display_name,
                            club.name AS club_name,
                            club.slug AS club_slug,
                            squad_player.fantasy_role,
                            squad_player.purchase_price

                        FROM fantasy_squad_players squad_player

                        JOIN players player
                            ON player.id =
                                squad_player.player_id

                        JOIN player_registrations registration
                            ON registration.player_id =
                                squad_player.player_id
                            AND registration.season_id =
                                squad_player.season_id
                            AND registration.registration_status =
                                'ACTIVE'

                        JOIN clubs club
                            ON club.id =
                                registration.club_id

                        WHERE squad_player.squad_id =
                            :squadId

                        ORDER BY
                            CASE squad_player.fantasy_role
                                WHEN 'P' THEN 1
                                WHEN 'D' THEN 2
                                WHEN 'C' THEN 3
                                WHEN 'A' THEN 4
                                ELSE 5
                            END,
                            player.display_name
                        """)
                        .param(
                                "squadId",
                                squad.id()
                        )
                        .query((rs, rowNumber) ->
                                new PlayerRow(
                                        rs.getLong(
                                                "player_id"
                                        ),
                                        rs.getString(
                                                "display_name"
                                        ),
                                        rs.getString(
                                                "club_name"
                                        ),
                                        rs.getString(
                                                "club_slug"
                                        ),
                                        rs.getString(
                                                "fantasy_role"
                                        ),
                                        rs.getInt(
                                                "purchase_price"
                                        )
                                )
                        )
                        .list();

        List<SquadCalendarPlayerResponse> response =
                new ArrayList<>();

        for (PlayerRow row : rows) {
            List<FixtureDifficultyResponse> fixtures =
                    fixtureDifficultyService
                            .findClubFixtures(
                                    squad.season(),
                                    row.clubSlug(),
                                    fromMatchday,
                                    toMatchday
                            );

            double averageDifficulty =
                    fixtures.stream()
                            .mapToDouble(fixture ->
                                    fixture
                                            .difficultyScore()
                                            .doubleValue()
                            )
                            .average()
                            .orElse(60.0);

            int easyFixtures =
                    countDifficulty(
                            fixtures,
                            "FACILE"
                    );

            int mediumFixtures =
                    countDifficulty(
                            fixtures,
                            "MEDIA"
                    );

            int difficultFixtures =
                    countDifficulty(
                            fixtures,
                            "DIFFICILE"
                    );

            if (fixtures.isEmpty()) {
                globalWarnings.add(
                        "Calendario non disponibile per "
                                + row.playerName()
                                + " - "
                                + row.club()
                );
            }

            response.add(
                    new SquadCalendarPlayerResponse(
                            row.playerId(),
                            row.playerName(),
                            row.club(),
                            row.clubSlug(),
                            row.role(),
                            row.purchasePrice(),
                            decimal(averageDifficulty),
                            easyFixtures,
                            mediumFixtures,
                            difficultFixtures,
                            List.copyOf(fixtures)
                    )
            );
        }

        return response;
    }

    private List<SquadPairingResponse> buildPairings(
            List<SquadCalendarPlayerResponse> players
    ) {
        List<PairingCandidate> candidates =
                new ArrayList<>();

        for (String role : ROLE_ORDER) {
            List<SquadCalendarPlayerResponse> rolePlayers =
                    players.stream()
                            .filter(player ->
                                    role.equals(
                                            player.role()
                                    )
                            )
                            .toList();

            for (int first = 0;
                    first < rolePlayers.size();
                    first++) {

                for (int second = first + 1;
                        second < rolePlayers.size();
                        second++) {

                    SquadCalendarPlayerResponse firstPlayer =
                            rolePlayers.get(first);

                    SquadCalendarPlayerResponse secondPlayer =
                            rolePlayers.get(second);

                    candidates.add(
                            createPairingCandidate(
                                    role,
                                    firstPlayer,
                                    secondPlayer
                            )
                    );
                }
            }
        }

        candidates.sort(
                Comparator
                        .comparing(
                                PairingCandidate::score,
                                Comparator.reverseOrder()
                        )
                        .thenComparing(
                                candidate ->
                                        candidate.firstPlayer()
                                                .playerName(),
                                String.CASE_INSENSITIVE_ORDER
                        )
                        .thenComparing(
                                candidate ->
                                        candidate.secondPlayer()
                                                .playerName(),
                                String.CASE_INSENSITIVE_ORDER
                        )
        );

        List<SquadPairingResponse> response =
                new ArrayList<>();

        int position = 1;

        for (PairingCandidate candidate : candidates) {
            response.add(
                    new SquadPairingResponse(
                            position++,
                            candidate.role(),
                            candidate.firstPlayer()
                                    .playerId(),
                            candidate.firstPlayer()
                                    .playerName(),
                            candidate.firstPlayer()
                                    .club(),
                            candidate.secondPlayer()
                                    .playerId(),
                            candidate.secondPlayer()
                                    .playerName(),
                            candidate.secondPlayer()
                                    .club(),
                            candidate.firstPlayer()
                                    .purchasePrice()
                                    + candidate.secondPlayer()
                                            .purchasePrice(),
                            decimal(candidate.score()),
                            pairingRecommendation(
                                    candidate.score()
                            ),
                            pairingExplanation(candidate)
                    )
            );
        }

        return response;
    }

    private PairingCandidate createPairingCandidate(
            String role,
            SquadCalendarPlayerResponse firstPlayer,
            SquadCalendarPlayerResponse secondPlayer
    ) {
        Map<Integer, Double> firstDifficulties =
                difficultiesByMatchday(
                        firstPlayer.fixtures()
                );

        Map<Integer, Double> secondDifficulties =
                difficultiesByMatchday(
                        secondPlayer.fixtures()
                );

        int coveredMatchdays = 0;
        int hardTogether = 0;
        double bestOptionTotal = 0;
        double differenceTotal = 0;

        for (Integer matchday :
                unionMatchdays(
                        firstDifficulties,
                        secondDifficulties
                )) {

            double firstDifficulty =
                    firstDifficulties.getOrDefault(
                            matchday,
                            60.0
                    );

            double secondDifficulty =
                    secondDifficulties.getOrDefault(
                            matchday,
                            60.0
                    );

            bestOptionTotal +=
                    Math.min(
                            firstDifficulty,
                            secondDifficulty
                    );

            differenceTotal +=
                    Math.abs(
                            firstDifficulty
                                    - secondDifficulty
                    );

            if (firstDifficulty >= 70
                    && secondDifficulty >= 70) {

                hardTogether++;
            }

            coveredMatchdays++;
        }

        double averageBestOption =
                coveredMatchdays == 0
                        ? 60.0
                        : bestOptionTotal
                                / coveredMatchdays;

        double averageDifference =
                coveredMatchdays == 0
                        ? 0
                        : differenceTotal
                                / coveredMatchdays;

        double score =
                100
                        - averageBestOption * 0.65
                        + Math.min(
                                25,
                                averageDifference * 0.70
                        )
                        - hardTogether * 6;

        if (!firstPlayer.club()
                .equalsIgnoreCase(
                        secondPlayer.club()
                )) {

            score += 5;
        }
        else {
            score -= 15;
        }

        score =
                clamp(score);

        return new PairingCandidate(
                role,
                firstPlayer,
                secondPlayer,
                score,
                averageBestOption,
                averageDifference,
                hardTogether
        );
    }

    private Map<Integer, Double> difficultiesByMatchday(
            List<FixtureDifficultyResponse> fixtures
    ) {
        Map<Integer, Double> response =
                new LinkedHashMap<>();

        for (FixtureDifficultyResponse fixture : fixtures) {
            response.put(
                    fixture.matchday(),
                    fixture.difficultyScore()
                            .doubleValue()
            );
        }

        return response;
    }

    private List<Integer> unionMatchdays(
            Map<Integer, Double> first,
            Map<Integer, Double> second
    ) {
        Map<Integer, Boolean> matchdays =
                new LinkedHashMap<>();

        first.keySet()
                .stream()
                .sorted()
                .forEach(matchday ->
                        matchdays.put(
                                matchday,
                                true
                        )
                );

        second.keySet()
                .stream()
                .sorted()
                .forEach(matchday ->
                        matchdays.put(
                                matchday,
                                true
                        )
                );

        return matchdays.keySet()
                .stream()
                .sorted()
                .toList();
    }

    private String pairingRecommendation(
            double score
    ) {
        if (score >= 70) {
            return "ABBINAMENTO ECCELLENTE";
        }

        if (score >= 55) {
            return "ABBINAMENTO CONSIGLIATO";
        }

        if (score >= 40) {
            return "ABBINAMENTO DISCRETO";
        }

        return "CALENDARI POCO COMPLEMENTARI";
    }

    private String pairingExplanation(
            PairingCandidate candidate
    ) {
        String clubText =
                candidate.firstPlayer()
                        .club()
                        .equalsIgnoreCase(
                                candidate.secondPlayer()
                                        .club()
                        )
                        ? "I giocatori appartengono allo stesso club."
                        : "I giocatori appartengono a club diversi.";

        return "Migliore difficoltÃ  media utilizzabile: "
                + decimal(
                        candidate.averageBestOption()
                )
                + ". Differenza media tra i calendari: "
                + decimal(
                        candidate.averageDifference()
                )
                + ". Giornate difficili contemporanee: "
                + candidate.hardTogether()
                + ". "
                + clubText;
    }

    private void addConcentrationWarnings(
            List<SquadCalendarPlayerResponse> players,
            List<String> globalWarnings
    ) {
        Map<String, Integer> totalByClub =
                new HashMap<>();

        Map<String, Integer> totalByClubAndRole =
                new HashMap<>();

        for (SquadCalendarPlayerResponse player : players) {
            totalByClub.merge(
                    player.club(),
                    1,
                    Integer::sum
            );

            totalByClubAndRole.merge(
                    player.club()
                            + "|"
                            + player.role(),
                    1,
                    Integer::sum
            );
        }

        totalByClub.forEach((club, total) -> {
            if (total >= 5) {
                globalWarnings.add(
                        "Concentrazione elevata: "
                                + total
                                + " giocatori appartengono al "
                                + club
                );
            }
        });

        totalByClubAndRole.forEach((key, total) -> {
            if (total >= 3) {
                String[] parts =
                        key.split(
                                "\\|",
                                2
                        );

                globalWarnings.add(
                        "Scarsa diversificazione nel ruolo "
                                + parts[1]
                                + ": "
                                + total
                                + " giocatori del "
                                + parts[0]
                );
            }
        });
    }

    private int countDifficulty(
            List<FixtureDifficultyResponse> fixtures,
            String difficulty
    ) {
        return (int) fixtures.stream()
                .filter(fixture ->
                        difficulty.equals(
                                fixture.difficultyLevel()
                        )
                )
                .count();
    }

    private String buildSummary(
            SquadResponse squad,
            int remainingSlots,
            boolean constraintsValid,
            boolean squadComplete,
            int pairings
    ) {
        String status =
                constraintsValid
                        ? "I vincoli attuali sono rispettati."
                        : "Sono presenti vincoli non rispettati.";

        if (squadComplete) {
            return "Rosa completa con "
                    + squad.totalPlayers()
                    + " giocatori, "
                    + squad.remainingBudget()
                    + " crediti residui e "
                    + pairings
                    + " abbinamenti analizzati. "
                    + status;
        }

        return "Rosa composta da "
                + squad.totalPlayers()
                + " giocatori su "
                + squad.maximumPlayers()
                + ". Restano "
                + remainingSlots
                + " slot e "
                + squad.remainingBudget()
                + " crediti. Sono stati analizzati "
                + pairings
                + " abbinamenti interni. "
                + status;
    }

    private void validateRequest(
            Long squadId,
            int fromMatchday,
            int toMatchday
    ) {
        if (squadId == null || squadId <= 0) {
            throw new BadRequestException(
                    "L'identificativo della rosa non Ã¨ valido"
            );
        }

        if (fromMatchday < 1
                || toMatchday > 38
                || fromMatchday > toMatchday) {

            throw new BadRequestException(
                    "L'intervallo delle giornate "
                            + "deve essere compreso tra 1 e 38"
            );
        }

        if (toMatchday - fromMatchday > 14) {
            throw new BadRequestException(
                    "L'analisi puÃ² comprendere al massimo "
                            + "15 giornate"
            );
        }
    }

    private BigDecimal decimal(
            double value
    ) {
        return BigDecimal
                .valueOf(value)
                .setScale(
                        2,
                        RoundingMode.HALF_UP
                );
    }

    private double clamp(
            double value
    ) {
        return Math.max(
                0,
                Math.min(
                        100,
                        value
                )
        );
    }
