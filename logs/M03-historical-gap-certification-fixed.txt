WITH historical AS (
    SELECT DISTINCT stats.player_id
    FROM player_season_statistics stats
    JOIN seasons s
        ON s.id = stats.season_id
    WHERE s.current_season = FALSE

    UNION

    SELECT DISTINCT r.player_id
    FROM player_roles r
    JOIN seasons s
        ON s.id = r.season_id
    WHERE s.current_season = FALSE
),
operational_catalog AS (
    SELECT DISTINCT p.id AS player_id
    FROM players p
    JOIN player_reconciliation_results rec
        ON rec.player_id = p.id
    JOIN seasons s
        ON s.id = rec.season_id
       AND s.current_season = TRUE
    JOIN player_registrations registration
        ON registration.player_id = p.id
       AND registration.season_id = s.id
    JOIN season_clubs sc
        ON sc.season_id = s.id
       AND sc.club_id = registration.club_id
    JOIN player_roles role
        ON role.player_id = p.id
       AND role.season_id = s.id
       AND role.source = 'FANTACALCIO'
    WHERE p.active = TRUE
      AND registration.registration_status = 'ACTIVE'
      AND role.fantasy_role IS NOT NULL
)
SELECT
    COUNT(*) AS historical_identities,
    COUNT(*) FILTER (
        WHERE oc.player_id IS NULL
    ) AS historical_not_operational_catalog,
    COUNT(*) FILTER (
        WHERE oc.player_id IS NOT NULL
    ) AS historical_still_operational
FROM historical h
LEFT JOIN operational_catalog oc
    ON oc.player_id = h.player_id;

WITH historical AS (
    SELECT DISTINCT
        stats.player_id,
        s.code AS season
    FROM player_season_statistics stats
    JOIN seasons s
        ON s.id = stats.season_id
    WHERE s.current_season = FALSE

    UNION

    SELECT DISTINCT
        r.player_id,
        s.code AS season
    FROM player_roles r
    JOIN seasons s
        ON s.id = r.season_id
    WHERE s.current_season = FALSE
),
operational_catalog AS (
    SELECT DISTINCT p.id AS player_id
    FROM players p
    JOIN player_reconciliation_results rec
        ON rec.player_id = p.id
    JOIN seasons s
        ON s.id = rec.season_id
       AND s.current_season = TRUE
    JOIN player_registrations registration
        ON registration.player_id = p.id
       AND registration.season_id = s.id
    JOIN season_clubs sc
        ON sc.season_id = s.id
       AND sc.club_id = registration.club_id
    JOIN player_roles role
        ON role.player_id = p.id
       AND role.season_id = s.id
       AND role.source = 'FANTACALCIO'
    WHERE p.active = TRUE
      AND registration.registration_status = 'ACTIVE'
      AND role.fantasy_role IS NOT NULL
)
SELECT
    p.id,
    p.display_name,
    p.active,
    STRING_AGG(
        DISTINCT h.season,
        ', '
        ORDER BY h.season
    ) AS seasons
FROM historical h
JOIN players p
    ON p.id = h.player_id
LEFT JOIN operational_catalog oc
    ON oc.player_id = p.id
WHERE oc.player_id IS NULL
GROUP BY
    p.id,
    p.display_name,
    p.active
ORDER BY p.display_name
LIMIT 50;
