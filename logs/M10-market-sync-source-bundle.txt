package it.fantacalciointelligence.backend.refresh;

    private final JdbcClient jdbcClient;

    private final PlayerReconciliationService
            reconciliationService;

    private final PlayerEvaluationGenerationService
            evaluationService;

    private final ClubStrengthGenerationService
            clubStrengthService;

    private final TransferDetectionService
            transferDetectionService;

    private final ReconciliationReviewService
            reconciliationReviewService;

    private final CatalogReadinessService
            catalogReadinessService;

    private final DataFreshnessService
            freshnessService;

    private final SelectiveIdentityEnrichmentService
            identityEnrichmentService;
    private final Duration staleThreshold;
    private final RefreshFaultInjector faultInjector;

    public DataRefreshService(
            JdbcClient jdbcClient,
            PlayerReconciliationService reconciliationService,
            PlayerEvaluationGenerationService evaluationService,
            ClubStrengthGenerationService clubStrengthService,
            TransferDetectionService transferDetectionService,
            ReconciliationReviewService reconciliationReviewService,
            CatalogReadinessService catalogReadinessService,
            DataFreshnessService freshnessService,
            SelectiveIdentityEnrichmentService identityEnrichmentService,
            RefreshFaultInjector faultInjector,
            @Value("${app.data-refresh.stale-threshold:2h}") Duration staleThreshold
    ) {
        this.jdbcClient = jdbcClient;
        this.reconciliationService =
                reconciliationService;
        this.evaluationService =
                evaluationService;
        this.clubStrengthService =
                clubStrengthService;
        this.transferDetectionService =
                transferDetectionService;
        this.reconciliationReviewService =
                reconciliationReviewService;
        this.catalogReadinessService =
                catalogReadinessService;
        this.freshnessService =
                freshnessService;
        this.identityEnrichmentService =
                identityEnrichmentService;
        this.faultInjector = faultInjector;
        if (staleThreshold == null || staleThreshold.isZero() || staleThreshold.isNegative()) {
            throw new IllegalStateException("app.data-refresh.stale-threshold deve essere maggiore di zero");
        }
        this.staleThreshold = staleThreshold;
    }

    public DataRefreshResponse runManual(
            String seasonCode
    ) {
        return run(
                seasonCode,
                "MANUAL"
        );
    }

    public DataRefreshResponse runScheduled(
            String seasonCode
    ) {
        return run(
                seasonCode,
                "SCHEDULED"
        );
    }

    public DataRefreshResponse findLatest(
            String seasonCode
    ) {

        return jdbcClient.sql("""
                SELECT
                    refresh.id,
                    season.code AS season,
                    refresh.trigger_type,
                    refresh.status,
                    refresh.source_freshness_status,
                    refresh.source_freshness_checked_at,
                    refresh.reconciliation_matched_players,
                    refresh.generated_evaluations,
                    refresh.generated_club_ratings,
                    refresh.started_at,
                    refresh.completed_at,
                    refresh.error_message

                FROM data_refresh_runs refresh

                JOIN seasons season
                    ON season.id =
                        refresh.season_id

                WHERE season.code = :season

                ORDER BY
                    refresh.started_at DESC,
                    refresh.id DESC

                LIMIT 1
                """)
                .param(
                        "season",
                        seasonCode
                )
                .query((resultSet, rowNumber) ->
                        mapResponse(resultSet)
                )
                .optional()
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Nessun aggiornamento trovato "
                                        + "per la stagione "
                                        + seasonCode
                        )
                );
    }

    private DataRefreshResponse run(
            String seasonCode,
            String triggerType
    ) {

        Long seasonId =
                findSeasonId(
                        seasonCode
                );

        Long refreshId =
                createRefreshRun(
                        seasonId,
                        triggerType
                );

        try {
            DataFreshnessResponse freshness =
                    freshnessService.evaluate(
                            seasonCode
                    );

            updateRefreshFreshness(
                    refreshId,
                    freshness.overallStatus()
            );

            if (freshness.blocked()) {
                throw new ConflictException(
                        "Dati sorgente non abbastanza aggiornati: "
                                + String.join(
                                        "; ",
                                        freshness.blockers()
                                )
                );
            }

            /*
             * Transfer detection MUST run against the previous
             * active registration before reconciliation applies
             * the newest Fantacalcio snapshot.
             */
            transferDetectionService.detectLatest(
                    seasonCode
            );
            faultInjector.checkpoint("AFTER_TRANSFER_DETECTION");

            ReconciliationResponse reconciliation =
                    reconciliationService.reconcile(
                            seasonCode
                    );
            faultInjector.checkpoint("AFTER_RECONCILIATION");

            /*
             * Persist unresolved identities after reconciliation
             * so an administrator has a durable review queue.
             */
            reconciliationReviewService.syncLatest(
                    seasonCode
            );
            faultInjector.checkpoint("AFTER_REVIEW_SYNC");

            /*
             * B71 production identity enrichment.
             *
             * It MUST run after reconciliation because Transfermarkt
             * discovery reads the newly reconciled operational catalog.
             *
             * Manual refreshes never initiate remote identity discovery.
             */
            if ("SCHEDULED".equals(triggerType)) {

                identityEnrichmentService.runScheduled(
                        seasonCode
                );

                faultInjector.checkpoint(
                        "AFTER_IDENTITY_ENRICHMENT"
                );
            }

            CatalogReadinessResponse readiness =
                    catalogReadinessService.evaluateLatest(
                            seasonCode
                    );

            /*
             * WARNING is allowed: unresolved identities and
             * pending transfers are expected during the market.
             *
             * BLOCKED means structural catalog corruption such
             * as reconciled players without a role or an ACTIVE
             * registration.
             */
            if (readiness.blocked()) {
                throw new ConflictException(
                        "Catalogo non pronto: "
                                + String.join(
                                        "; ",
                                        readiness.blockers()
                                )
                );
            }

            PlayerEvaluationGenerationResponse evaluations =
                    evaluationService.generate(
                            seasonCode
                    );
            faultInjector.checkpoint("AFTER_EVALUATIONS");

            ClubStrengthGenerationResponse clubRatings =
                    clubStrengthService.generate(
                            seasonCode
                    );
            faultInjector.checkpoint("AFTER_CLUB_STRENGTH");
            faultInjector.checkpoint("BEFORE_COMPLETE");

            completeRefreshRun(
                    refreshId,
                    reconciliation.matchedPlayers(),
                    evaluations.generatedEvaluations(),
                    clubRatings.generatedRatings()
            );

            return findById(refreshId);

        } catch (RuntimeException exception) {

            markRefreshRunFailed(
                    refreshId,
                    readableMessage(exception)
            );

            throw exception;
        }
    }

    private Long findSeasonId(
            String seasonCode
    ) {

        return jdbcClient.sql("""
                SELECT id
                FROM seasons
                WHERE code = :season
                """)
                .param(
                        "season",
                        seasonCode
                )
                .query(Long.class)
                .optional()
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Stagione non trovata: "
                                        + seasonCode
                        )
                );
    }

    Long createRefreshRun(
            Long seasonId,
            String triggerType
    ) {

        recoverStaleRefresh(seasonId);
        try {
            return jdbcClient.sql("""
                    INSERT INTO data_refresh_runs (
                        season_id,
                        trigger_type,
                        status
                    )
                    VALUES (
                        :seasonId,
                        :triggerType,
                        'RUNNING'
                    )
                    RETURNING id
                    """)
                    .param(
                            "seasonId",
                            seasonId
                    )
                    .param(
                            "triggerType",
                            triggerType
                    )
                    .query(Long.class)
                    .single();

        } catch (DataIntegrityViolationException exception) {

            throw new ConflictException(
                    "Un aggiornamento dati è già in esecuzione"
            );
        }
    }

    private void recoverStaleRefresh(Long seasonId) {
        jdbcClient.sql("""
                UPDATE data_refresh_runs
                SET status = 'FAILED', completed_at = CURRENT_TIMESTAMP,
                    error_message = 'Esecuzione interrotta recuperata automaticamente'
                WHERE season_id = :seasonId
                  AND status = 'RUNNING'
                  AND started_at < CURRENT_TIMESTAMP - (:staleSeconds * INTERVAL '1 second')
                """)
                .param("seasonId", seasonId)
                .param("staleSeconds", staleThreshold.toSeconds())
                .update();
    }

    private void updateRefreshFreshness(
            Long refreshId,
            String freshnessStatus
    ) {

        jdbcClient.sql("""
                UPDATE data_refresh_runs
                SET
                    source_freshness_status =
                        :freshnessStatus,

                    source_freshness_checked_at =
                        CURRENT_TIMESTAMP

                WHERE id = :refreshId
                """)
                .param(
                        "freshnessStatus",
                        freshnessStatus
                )
                .param(
                        "refreshId",
                        refreshId
                )
                .update();
    }

    private void completeRefreshRun(
            Long refreshId,
            int matchedPlayers,
            int generatedEvaluations,
            int generatedClubRatings
    ) {

        jdbcClient.sql("""
                UPDATE data_refresh_runs
                SET
                    status = 'COMPLETED',

                    reconciliation_matched_players =
                        :matchedPlayers,

                    generated_evaluations =
                        :generatedEvaluations,

                    generated_club_ratings =
                        :generatedClubRatings,

                    completed_at =
                        CURRENT_TIMESTAMP,

                    error_message =
                        NULL

                WHERE id = :refreshId
                """)
                .param(
                        "matchedPlayers",
                        matchedPlayers
                )
                .param(
                        "generatedEvaluations",
                        generatedEvaluations
                )
                .param(
                        "generatedClubRatings",
                        generatedClubRatings
                )
                .param(
                        "refreshId",
                        refreshId
                )
                .update();
    }

    private void markRefreshRunFailed(
            Long refreshId,
            String errorMessage
    ) {

        jdbcClient.sql("""
                UPDATE data_refresh_runs
                SET
                    status = 'FAILED',

                    completed_at =
                        CURRENT_TIMESTAMP,

                    error_message =
                        :errorMessage

                WHERE id = :refreshId
                """)
                .param(
                        "errorMessage",
                        truncate(
                                errorMessage,
                                2000
                        ),
                        Types.VARCHAR
                )
                .param(
                        "refreshId",
                        refreshId
                )
                .update();
    }

    private DataRefreshResponse findById(
            Long refreshId
    ) {

        return jdbcClient.sql("""
                SELECT
                    refresh.id,
                    season.code AS season,
                    refresh.trigger_type,
                    refresh.status,
                    refresh.source_freshness_status,
                    refresh.source_freshness_checked_at,
                    refresh.reconciliation_matched_players,
                    refresh.generated_evaluations,
                    refresh.generated_club_ratings,
                    refresh.started_at,
                    refresh.completed_at,
                    refresh.error_message

                FROM data_refresh_runs refresh

                JOIN seasons season
                    ON season.id =
                        refresh.season_id

                WHERE refresh.id = :refreshId
                """)
                .param(
                        "refreshId",
                        refreshId
                )
                .query((resultSet, rowNumber) ->
                        mapResponse(resultSet)
                )
                .single();
    }

    private DataRefreshResponse mapResponse(
            java.sql.ResultSet resultSet
    ) throws java.sql.SQLException {

        return new DataRefreshResponse(
                resultSet.getLong("id"),
                resultSet.getString("season"),

                resultSet.getString(
                        "trigger_type"
                ),

                resultSet.getString("status"),

                resultSet.getString(
                        "source_freshness_status"
                ),

                resultSet.getObject(
                        "source_freshness_checked_at",
                        OffsetDateTime.class
                ),

                resultSet.getInt(
                        "reconciliation_matched_players"
                ),

                resultSet.getInt(
                        "generated_evaluations"
                ),

                resultSet.getInt(
                        "generated_club_ratings"
                ),

                resultSet.getObject(
                        "started_at",
                        OffsetDateTime.class
                ),

                resultSet.getObject(
                        "completed_at",
                        OffsetDateTime.class
                ),

                resultSet.getString(
                        "error_message"
                )
        );
    }

    private String readableMessage(
            Throwable exception
    ) {

        return SecretSanitizer.sanitize(exception);
    }

    private String truncate(
            String value,
            int maximumLength
    ) {

        if (value == null) {
            return null;
        }

        if (value.length()
                <= maximumLength) {

            return value;
        }

        return value.substring(
                0,
                maximumLength
        );
    }
}

package it.fantacalciointelligence.backend.ingestion;

import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import it.fantacalciointelligence.backend.common.ConflictException;
import it.fantacalciointelligence.backend.refresh.DataRefreshResponse;
import it.fantacalciointelligence.backend.refresh.DataRefreshService;

@Service
public class AutomatedDataUpdateService {

    private final FantacalcioListoneProperties listoneProperties;
    private final FantacalcioListoneSyncService listoneSyncService;
    private final SourceIngestionProperties sourceIngestionProperties;
    private final SourceIngestionService sourceIngestionService;
    private final DataRefreshService refreshService;
    private final AutomationRunService automationRunService;

    @Autowired
    public AutomatedDataUpdateService(
            FantacalcioListoneProperties listoneProperties,
            FantacalcioListoneSyncService listoneSyncService,
            SourceIngestionProperties sourceIngestionProperties,
            SourceIngestionService sourceIngestionService,
            DataRefreshService refreshService,
            AutomationRunService automationRunService
    ) {
        this.listoneProperties = listoneProperties;
        this.listoneSyncService = listoneSyncService;
        this.sourceIngestionProperties = sourceIngestionProperties;
        this.sourceIngestionService = sourceIngestionService;
        this.refreshService = refreshService;
        this.automationRunService = automationRunService;
    }

    /*
     * Compatibility constructor used by existing focused unit tests.
     *
     * It intentionally preserves the historical isolated service-test path.
     * Spring production wiring always uses the @Autowired constructor above,
     * therefore real scheduled executions always use AutomationRunService.
     */
    AutomatedDataUpdateService(
            SourceIngestionProperties sourceIngestionProperties,
            SourceIngestionService sourceIngestionService,
            DataRefreshService refreshService
    ) {

        FantacalcioListoneProperties disabledListone =
                new FantacalcioListoneProperties();

        disabledListone.setEnabled(false);

        this.listoneProperties = disabledListone;
        this.listoneSyncService = null;
        this.sourceIngestionProperties = sourceIngestionProperties;
        this.sourceIngestionService = sourceIngestionService;
        this.refreshService = refreshService;
        this.automationRunService = null;
    }

    public AutomatedUpdateResponse run(
            String season
    ) {

        if (automationRunService == null) {
            return runPipeline(
                    season,
                    null
            );
        }

        long automationRunId =
                automationRunService.begin(
                        season
                );

        try {

            AutomatedUpdateResponse result =
                    runPipeline(
                            season,
                            automationRunId
                    );

            automationRunService.complete(
                    automationRunId,
                    result.listone() == null
                            ? null
                            : result.listone().runId(),
                    result.ingestion().size(),
                    result.refresh() == null
                            ? null
                            : result.refresh().id()
            );

            return result;

        } catch (RuntimeException exception) {

            automationRunService.fail(
                    automationRunId,
                    exception
            );

            throw exception;
        }
    }

    private AutomatedUpdateResponse runPipeline(
            String season,
            Long automationRunId
    ) {

        FantacalcioListoneSyncResponse listone =
                null;

        if (listoneProperties.isEnabled()) {

            heartbeat(
                    automationRunId,
                    "LISTONE"
            );

            if (listoneSyncService == null) {
                throw new IllegalStateException(
                        "Fantacalcio listone sync abilitato senza servizio configurato"
                );
            }

            listone =
                    listoneSyncService.sync(
                            season
                    );
        }

        List<SourceIngestionResponse> ingestion =
                List.of();

        if (sourceIngestionProperties.isEnabled()) {

            heartbeat(
                    automationRunId,
                    "INGESTION"
            );

            ingestion =
                    sourceIngestionService.ingestAll(
                            season
                    );

            boolean requiredFailed =
                    ingestion.stream()
                            .anyMatch(response ->
                                    response.required()
                                            && "FAILED".equals(
                                                    response.status()
                                            )
                            );

            if (requiredFailed) {
                throw new ConflictException(
                        "Ingestion richiesta fallita; refresh non eseguito"
                );
            }
        }

        heartbeat(
                automationRunId,
                "REFRESH"
        );

        DataRefreshResponse refresh =
                refreshService.runScheduled(
                        season
                );

        return new AutomatedUpdateResponse(
                listone,
                ingestion,
                refresh
        );
    }

    private void heartbeat(
            Long automationRunId,
            String stage
    ) {

        if (
                automationRunId != null
                        && automationRunService != null
        ) {
            automationRunService.stage(
                    automationRunId,
                    stage
            );
        }
    }

    public record AutomatedUpdateResponse(
            FantacalcioListoneSyncResponse listone,
            List<SourceIngestionResponse> ingestion,
            DataRefreshResponse refresh
    ) {

        public AutomatedUpdateResponse(
                List<SourceIngestionResponse> ingestion,
                DataRefreshResponse refresh
        ) {
            this(
                    null,
                    ingestion,
                    refresh
            );
        }
    }
}

package it.fantacalciointelligence.backend.ingestion;

import java.io.ByteArrayOutputStream;
import java.net.InetAddress;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.nio.charset.StandardCharsets;
import java.util.Locale;

import org.springframework.stereotype.Component;

import it.fantacalciointelligence.backend.common.BadRequestException;

@Component
public class FantacalcioListoneClient {

    private final FantacalcioListoneProperties properties;
    private final HttpClient client;

    public FantacalcioListoneClient(FantacalcioListoneProperties properties) {
        this.properties = properties;
        this.client = HttpClient.newBuilder()
                .connectTimeout(properties.getConnectTimeout())
                .followRedirects(HttpClient.Redirect.NEVER)
                .build();
    }

    public FetchResult fetch() {
        properties.validate();

        URI uri = validate(properties.getUrl());
        int redirects = 0;

        try {
            while (true) {
                HttpRequest request = HttpRequest.newBuilder(uri)
                        .timeout(properties.getRequestTimeout())
                        .header(
                                "User-Agent",
                                "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                                        + "AppleWebKit/537.36 Chrome/151 Safari/537.36"
                        )
                        .header("Accept", "text/html,application/xhtml+xml;q=0.9,*/*;q=0.1")
                        .GET()
                        .build();

                HttpResponse<java.io.InputStream> response = client.send(
                        request,
                        HttpResponse.BodyHandlers.ofInputStream()
                );

                int status = response.statusCode();

                if (status >= 300 && status < 400) {
                    String location;
                    try (var body = response.body()) {
                        if (++redirects > properties.getMaxRedirects()) {
                            throw new BadRequestException("Troppi redirect Fantacalcio");
                        }
                        location = response.headers()
                                .firstValue("location")
                                .orElseThrow(() -> new BadRequestException(
                                        "Redirect Fantacalcio senza destinazione"
                                ));
                    }
                    uri = validate(uri.resolve(location).toString());
                    continue;
                }

                if (status < 200 || status >= 300) {
                    response.body().close();
                    throw new BadRequestException(
                            "Fantacalcio listone ha risposto HTTP " + status
                    );
                }

                String contentType = response.headers()
                        .firstValue("content-type")
                        .orElse("")
                        .toLowerCase(Locale.ROOT);

                if (!contentType.contains("text/html")) {
                    response.body().close();
                    throw new BadRequestException(
                            "Fantacalcio listone non ha restituito HTML"
                    );
                }

                try (
                        var input = response.body();
                        var output = new ByteArrayOutputStream()
                ) {
                    byte[] buffer = new byte[8192];
                    int total = 0;
                    int read;

                    while ((read = input.read(buffer)) != -1) {
                        total += read;
                        if (total > properties.getMaxBytes()) {
                            throw new BadRequestException(
                                    "HTML Fantacalcio supera il limite configurato"
                            );
                        }
                        output.write(buffer, 0, read);
                    }

                    if (total == 0) {
                        throw new BadRequestException("HTML Fantacalcio vuoto");
                    }

                    return new FetchResult(
                            output.toString(StandardCharsets.UTF_8),
                            status,
                            uri,
                            total
                    );
                }
            }
        } catch (BadRequestException exception) {
            throw exception;
        } catch (Exception exception) {
            throw new BadRequestException(
                    "Download listone Fantacalcio fallito",
                    exception
            );
        }
    }

    URI validate(String value) {
        try {
            URI uri = URI.create(value);

            if (!uri.isAbsolute()
                    || uri.getHost() == null
                    || uri.getUserInfo() != null) {
                throw new BadRequestException("URL listone Fantacalcio non valido");
            }

            if (!"https".equalsIgnoreCase(uri.getScheme())) {
                throw new BadRequestException(
                        "Il listone Fantacalcio richiede HTTPS"
                );
            }

            String host = uri.getHost().toLowerCase(Locale.ROOT);
            if (!host.equals("fantacalcio.it")
                    && !host.endsWith(".fantacalcio.it")) {
                throw new BadRequestException(
                        "Host listone Fantacalcio non consentito"
                );
            }

            for (InetAddress address : InetAddress.getAllByName(uri.getHost())) {
                if (address.isAnyLocalAddress()
                        || address.isLoopbackAddress()
                        || address.isLinkLocalAddress()
                        || address.isSiteLocalAddress()) {
                    throw new BadRequestException(
                            "Host privato/locale non consentito"
                    );
                }
            }

            return uri;
        } catch (BadRequestException exception) {
            throw exception;
        } catch (Exception exception) {
            throw new BadRequestException(
                    "URL listone Fantacalcio non valido",
                    exception
            );
        }
    }

    public record FetchResult(
            String html,
            int httpStatus,
            URI finalUri,
            long payloadBytes
    ) {
    }
}

import org.springframework.stereotype.Component;

@Component
@ConfigurationProperties(prefix = "app.fantacalcio-listone")
public class FantacalcioListoneProperties {

import java.io.ByteArrayOutputStream;
import java.io.OutputStreamWriter;

@Service
public class FantacalcioListoneSyncService {

    static final String FEED_ID = "fantacalcio-listone-html";
    static final String SOURCE = "FANTACALCIO";

    private final FantacalcioListoneProperties properties;
    private final FantacalcioListoneClient client;
    private final FantacalcioListoneParser parser;
    private final PlayerImportService playerImportService;
    private final JdbcClient jdbc;

    public FantacalcioListoneSyncService(
            FantacalcioListoneProperties properties,
            FantacalcioListoneClient client,
            FantacalcioListoneParser parser,
            PlayerImportService playerImportService,
            JdbcClient jdbc
    ) {
        this.properties = properties;
        this.client = client;
        this.parser = parser;
        this.playerImportService = playerImportService;
        this.jdbc = jdbc;
    }

    public FantacalcioListoneSyncResponse sync(String season) {
        if (!properties.isEnabled()) {
            throw new ConflictException("Fantacalcio listone sync disabilitato");
        }

        long seasonId = seasonId(season);
        long runId = start(seasonId);

        try {
            FantacalcioListoneClient.FetchResult fetched = client.fetch();
            List<FantacalcioListoneParser.ListonePlayer> parsed =
                    parser.parse(fetched.html());

            Map<String, String> clubNamesBySlug = currentSeasonClubs(seasonId);
            byte[] csv = toCsv(parsed, clubNamesBySlug);
            String checksum = sha256(csv);

            Optional<String> previous = latestEffectiveChecksum(
                    seasonId,
                    runId
            );

            if (previous.filter(checksum::equals).isPresent()) {
                finish(
                        runId,
                        "SKIPPED_UNCHANGED",
                        fetched.httpStatus(),
                        fetched.payloadBytes(),
                        checksum,
                        null,
                        null
                );

                return response(runId, season, parsed.size());
            }

            String filename = "players-fantacalcio-" + season.replace("/", "-") + ".csv";
            PlayerImportResponse imported = playerImportService.importPayload(
                    SOURCE,
                    season,
                    new CsvImportPayload(filename, csv)
            );

            jdbc.sql("""
                    UPDATE import_batches
                    SET source_observed_at = CURRENT_TIMESTAMP
                    WHERE id = :id
                    """)
                    .param("id", imported.batchId())
                    .update();

            finish(
                    runId,
                    "COMPLETED",
                    fetched.httpStatus(),
                    fetched.payloadBytes(),
                    checksum,
                    imported.batchId(),
                    null
            );

            return response(runId, season, parsed.size());
        } catch (RuntimeException exception) {
            finish(
                    runId,
                    "FAILED",
                    null,
                    null,
                    null,
                    null,
                    SecretSanitizer.sanitize(exception)
            );
            throw exception;
        }
    }

    private byte[] toCsv(
            List<FantacalcioListoneParser.ListonePlayer> players,
            Map<String, String> clubNamesBySlug
    ) {
        try {
            ByteArrayOutputStream output = new ByteArrayOutputStream();

            try (
                    OutputStreamWriter writer = new OutputStreamWriter(
                            output,
                            StandardCharsets.UTF_8
                    );
                    CSVPrinter printer = CSVFormat.DEFAULT.builder()
                            .setHeader(
                                    "external_id",
                                    "name",
                                    "club",
                                    "role",
                                    "quotation",
                                    "fmv",
                                    "out_of_game"
                            )
                            .get()
                            .print(writer)
            ) {
                for (FantacalcioListoneParser.ListonePlayer player : players) {
                    String club = clubNamesBySlug.get(player.clubSlug());
                    if (club == null) {
                        throw new BadRequestException(
                                "Club Fantacalcio non riconosciuto per la stagione: "
                                        + player.clubSlug()
                        );
                    }

                    printer.printRecord(
                            player.externalId(),
                            player.name(),
                            club,
                            player.role(),
                            number(player.quotation()),
                            number(player.fmv()),
                            Boolean.toString(player.outOfGame())
                    );
                }
            }

            return output.toByteArray();
        } catch (BadRequestException exception) {
            throw exception;
        } catch (Exception exception) {
            throw new BadRequestException(
                    "Conversione listone Fantacalcio in CSV fallita",
                    exception
            );
        }
    }

    private Object number(BigDecimal value) {
        return value == null ? "" : value.stripTrailingZeros().toPlainString();
    }

    private Map<String, String> currentSeasonClubs(long seasonId) {
        List<String> clubs = jdbc.sql("""
                SELECT club.name
                FROM season_clubs membership
                JOIN clubs club ON club.id = membership.club_id
                WHERE membership.season_id = :seasonId
                """)
                .param("seasonId", seasonId)
                .query(String.class)
                .list();

        Map<String, String> result = new HashMap<>();
        for (String club : clubs) {
            result.put(slugify(club), club);
        }

        // Fantacalcio URL conventions / common aliases.
        alias(result, "inter", "Internazionale");
        alias(result, "milan", "Milan");
        alias(result, "roma", "Roma");
        alias(result, "lazio", "Lazio");

        return result;
    }

    private void alias(Map<String, String> map, String slug, String expectedName) {
        if (map.containsKey(slug)) {
            return;
        }
        map.values().stream()
                .filter(value -> value.equalsIgnoreCase(expectedName))
                .findFirst()
                .ifPresent(value -> map.put(slug, value));
    }

    static String slugify(String value) {
        String normalized = java.text.Normalizer
                .normalize(value, java.text.Normalizer.Form.NFD)
                .replaceAll("\\p{M}", "")
                .toLowerCase(Locale.ROOT)
                .replace("&", " ")
                .replaceAll("[^a-z0-9]+", "-")
                .replaceAll("^-+|-+$", "");
        return normalized;
    }

    private long start(long seasonId) {
        recoverStale(seasonId);
        try {
            return jdbc.sql("""
                    INSERT INTO source_ingestion_runs (
                        feed_id,
                        feed_type,
                        source,
                        season_id,
                        sanitized_url,
                        status
                    )
                    VALUES (
                        :feed,
                        'PLAYERS',
                        :source,
                        :season,
                        :url,
                        'STARTED'
                    )
                    RETURNING id
                    """)
                    .param("feed", FEED_ID)
                    .param("source", SOURCE)
                    .param("season", seasonId)
                    .param("url", properties.getUrl())
                    .query(Long.class)
                    .single();
        } catch (DataIntegrityViolationException exception) {
            throw new ConflictException(
                    "Sync listone Fantacalcio gia' in esecuzione"
            );
        }
    }

    private void recoverStale(long seasonId) {
        jdbc.sql("""
                UPDATE source_ingestion_runs
                SET status = 'FAILED',
                    completed_at = CURRENT_TIMESTAMP,
                    error_message = 'Sync listone interrotta recuperata automaticamente'
                WHERE feed_id = :feed
                  AND season_id = :season
                  AND status = 'STARTED'
                  AND started_at <
                      CURRENT_TIMESTAMP
                      - (:staleSeconds * INTERVAL '1 second')
                """)
                .param("feed", FEED_ID)
                .param("season", seasonId)
                .param(
                        "staleSeconds",
                        properties.getStaleThreshold().toSeconds()
                )
                .update();
    }

    private Optional<String> latestEffectiveChecksum(
            long seasonId,
            long runId
    ) {
        return jdbc.sql("""
                SELECT payload_sha256
                FROM source_ingestion_runs
                WHERE feed_id = :feed
                  AND season_id = :season
                  AND status IN ('COMPLETED', 'SKIPPED_UNCHANGED')
                  AND id < :runId
                ORDER BY id DESC
                LIMIT 1
                """)
                .param("feed", FEED_ID)
                .param("season", seasonId)
                .param("runId", runId)
                .query(String.class)
                .optional();
    }

    private void finish(
            long runId,
            String status,
            Integer httpStatus,
            Long payloadBytes,
            String checksum,
            Long importBatchId,
            String error
    ) {
        jdbc.sql("""
                UPDATE source_ingestion_runs
                SET status = :status,
                    http_status = :httpStatus,
                    payload_bytes = :payloadBytes,
                    payload_sha256 = :checksum,
                    import_batch_id = :importBatchId,
                    error_message = :error,
                    completed_at = CURRENT_TIMESTAMP
                WHERE id = :runId
                """)
                .param("status", status)
                .param("httpStatus", httpStatus, java.sql.Types.INTEGER)
                .param("payloadBytes", payloadBytes, java.sql.Types.BIGINT)
                .param("checksum", checksum, java.sql.Types.VARCHAR)
                .param("importBatchId", importBatchId, java.sql.Types.BIGINT)
                .param(
                        "error",
                        error == null ? null : truncate(error, 2000),
                        java.sql.Types.VARCHAR
                )
                .param("runId", runId)
                .update();
    }

    private FantacalcioListoneSyncResponse response(
            long runId,
            String season,
            int parsedPlayers
    ) {
        return jdbc.sql("""
                SELECT id,
                       status,
                       payload_sha256,
                       import_batch_id,
                       started_at,
                       completed_at
                FROM source_ingestion_runs
                WHERE id = :id
                """)
                .param("id", runId)
                .query((rs, rowNum) -> mapResponse(
                        rs,
                        season,
                        parsedPlayers
                ))
                .single();
    }

    private FantacalcioListoneSyncResponse mapResponse(
            ResultSet rs,
            String season,
            int parsedPlayers
    ) throws SQLException {
        return new FantacalcioListoneSyncResponse(
                rs.getLong("id"),
                season,
                rs.getString("status"),
                parsedPlayers,
                (Long) rs.getObject("import_batch_id"),
                rs.getString("payload_sha256"),
                rs.getObject("started_at", OffsetDateTime.class),
                rs.getObject("completed_at", OffsetDateTime.class)
        );
    }

    private long seasonId(String code) {
        return jdbc.sql("""
                SELECT id
                FROM seasons
                WHERE code = :code
                """)
                .param("code", code)
                .query(Long.class)
                .optional()
                .orElseThrow(() -> new ResourceNotFoundException(
                        "Stagione non trovata: " + code
                ));
    }

    private String sha256(byte[] value) {
        try {
            return HexFormat.of().formatHex(
                    MessageDigest.getInstance("SHA-256").digest(value)
            );
        } catch (Exception exception) {
            throw new IllegalStateException(exception);
        }
    }

    private String truncate(String value, int max) {
        return value.length() <= max ? value : value.substring(0, max);
    }

package it.fantacalciointelligence.backend.ingestion;

import java.io.ByteArrayOutputStream;
import java.net.InetAddress;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.util.Locale;

import org.springframework.stereotype.Component;

import it.fantacalciointelligence.backend.common.BadRequestException;

@Component
public class RemoteCsvClient {

    private final SourceIngestionProperties config;
    private final HttpClient client;

    public RemoteCsvClient(SourceIngestionProperties config) {
        this.config = config;
        this.client = HttpClient.newBuilder()
                .connectTimeout(config.getConnectTimeout())
                .followRedirects(HttpClient.Redirect.NEVER)
                .build();
    }

    public FetchResult fetch(String value) {
        URI uri = validate(value);
        int redirects = 0;
        try {
            while (true) {
                HttpRequest request = HttpRequest.newBuilder(uri)
                        .timeout(config.getRequestTimeout())
                        .header("User-Agent", "Fantacalcio-Intelligence-Source-Ingestion/1.0")
                        .header("Accept", "text/csv,text/plain;q=0.9,*/*;q=0.1")
                        .GET()
                        .build();
                HttpResponse<java.io.InputStream> response = client.send(
                        request,
                        HttpResponse.BodyHandlers.ofInputStream()
                );
                int status = response.statusCode();
                if (status >= 300 && status < 400) {
                    String location;
                    try (var ignored = response.body()) {
                        if (++redirects > config.getMaxRedirects()) {
                            throw new BadRequestException("Troppi redirect remoti");
                        }
                        location = response.headers().firstValue("location")
                                .orElseThrow(() -> new BadRequestException(
                                        "Redirect remoto senza destinazione"
                                ));
                    }
                    uri = validate(uri.resolve(location).toString());
                    continue;
                }
                if (status < 200 || status >= 300) {
                    throw new BadRequestException("Fonte remota ha risposto HTTP " + status);
                }
                String contentType = response.headers().firstValue("content-type")
                        .orElse("").toLowerCase(Locale.ROOT);
                if (contentType.startsWith("text/html")) {
                    response.body().close();
                    throw new BadRequestException("La fonte remota ha restituito HTML invece di CSV");
                }
                try (var input = response.body(); var output = new ByteArrayOutputStream()) {
                    byte[] buffer = new byte[8192];
                    int total = 0;
                    int read;
                    while ((read = input.read(buffer)) != -1) {
                        total += read;
                        if (total > config.getMaxBytes()) {
                            throw new BadRequestException("Payload remoto supera il limite configurato");
                        }
                        output.write(buffer, 0, read);
                    }
                    if (total == 0) {
                        throw new BadRequestException("Payload remoto vuoto");
                    }
                    return new FetchResult(output.toByteArray(), status, uri, safeFilename(uri));
                }
            }
        } catch (BadRequestException exception) {
            throw exception;
        } catch (Exception exception) {
            throw new BadRequestException("Download remoto fallito", exception);
        }
    }

    URI validate(String value) {
        try {
            URI uri = URI.create(value);
            if (!uri.isAbsolute() || uri.getHost() == null || uri.getUserInfo() != null) {
                throw new BadRequestException("URL feed non valido");
            }
            String scheme = uri.getScheme().toLowerCase(Locale.ROOT);
            if (!scheme.equals("https") && !(scheme.equals("http") && config.isAllowHttp())) {
                throw new BadRequestException("Schema URL feed non consentito");
            }
            if (!config.isAllowPrivateHosts()) {
                for (InetAddress address : InetAddress.getAllByName(uri.getHost())) {
                    if (address.isAnyLocalAddress()
                            || address.isLoopbackAddress()
                            || address.isLinkLocalAddress()
                            || address.isSiteLocalAddress()) {
                        throw new BadRequestException("Host privato/locale non consentito");
                    }
                }
            }
            return uri;
        } catch (BadRequestException exception) {
            throw exception;
        } catch (Exception exception) {
            throw new BadRequestException("URL feed non valido", exception);
        }
    }

    public static String sanitized(URI uri) {
        try {
            return new URI(
                    uri.getScheme(), null, uri.getHost(), uri.getPort(), uri.getPath(), null, null
            ).toString();
        } catch (Exception exception) {
            return uri.getScheme() + "://" + uri.getHost();
        }
    }

    private String safeFilename(URI uri) {
        String path = uri.getPath();
        String name = path == null ? "" : path.substring(path.lastIndexOf('/') + 1);
        return name.matches("[A-Za-z0-9._-]{1,255}") ? name : "remote-feed.csv";
    }

    public record FetchResult(byte[] bytes, int httpStatus, URI finalUri, String filename) {}
}

                SourceFeedType.valueOf(feed.type.trim().toUpperCase(Locale.ROOT));

                throw invalid("tipo feed non valido: " + feed.type);

            requireText(feed.source, "source");
            requireText(feed.season, "season");
            requireText(feed.url, "url");

                    && (feed.filename.isBlank()

    public List<SourceIngestionResponse> latest(String season) {
        return jdbc.sql(baseSelect() + """
                JOIN seasons season ON season.id = run.season_id
                WHERE season.code = :season
                  AND run.id IN (

    private SourceIngestionResponse ingest(SourceIngestionProperties.Feed feed) {
        validateFeed(feed);
        long seasonId = seasonId(feed.getSeason());
        long runId = start(feed, seasonId);
        try {
            RemoteCsvClient.FetchResult fetched = remote.fetch(feed.getUrl());
            String checksum = sha256(fetched.bytes());
            Optional<String> effectiveChecksum = latestEffectiveChecksum(
                    feed.getId(), seasonId, runId
            );
            if (effectiveChecksum.filter(checksum::equals).isPresent()) {
                finish(
                        runId,
                        "SKIPPED_UNCHANGED",
                        fetched.httpStatus(),
                        (long) fetched.bytes().length,
                        checksum,
                        null,
                        null
                );
                return find(runId, feed.isRequired());
            }

            String filename = feed.getFilename() == null || feed.getFilename().isBlank()
                    ? fetched.filename()
                    : feed.getFilename();
            CsvImportPayload payload = new CsvImportPayload(filename, fetched.bytes());
            SourceFeedType feedType = type(feed);
            long batchId = importPayload(feedType, feed, payload);
            observe(feedType, batchId);
            finish(
                    runId,
                    "COMPLETED",
                    fetched.httpStatus(),
                    (long) fetched.bytes().length,
                    checksum,
                    batchId,
                    null
            );
            LOG.info(
                    "Source ingestion feed={} source={} season={} status=COMPLETED "
                            + "bytes={} checksum={} batch={}",
                    feed.getId(),
                    feed.getSource(),
                    feed.getSeason(),
                    fetched.bytes().length,
                    checksum.substring(0, 12),
                    batchId
            );
            return find(runId, feed.isRequired());
        } catch (RuntimeException exception) {
            finish(runId, "FAILED", null, null, null, null, safe(exception));
            LOG.warn(
                    "Source ingestion feed={} source={} season={} status=FAILED",
                    feed.getId(),
                    feed.getSource(),
                    feed.getSeason()
            );
            return find(runId, feed.isRequired());
        }
    }

    private Optional<String> latestEffectiveChecksum(String feedId, long seasonId, long runId) {
        return jdbc.sql("""
                SELECT payload_sha256
                FROM source_ingestion_runs
                WHERE feed_id = :feed
                  AND season_id = :season
                  AND status IN ('COMPLETED', 'SKIPPED_UNCHANGED')
                  AND id < :run
                ORDER BY id DESC
                LIMIT 1
                """)
                .param("feed", feedId)
                .param("season", seasonId)
                .param("run", runId)
                .query(String.class)
                .optional();
    }

    private long importPayload(
            SourceFeedType feedType,
            SourceIngestionProperties.Feed feed,
            CsvImportPayload payload
    ) {
        return switch (feedType) {
            case PLAYERS -> players.importPayload(
                    feed.getSource(), feed.getSeason(), payload
            ).batchId();
            case STATISTICS -> statistics.importPayload(
                    feed.getSource(), feed.getSeason(), payload
            ).batchId();
            case FIXTURES -> fixtures.importPayload(
                    feed.getSource(), feed.getSeason(), payload
            ).batchId();
        };
    }

    long start(SourceIngestionProperties.Feed feed, long seasonId) {
        jdbc.sql("""
                UPDATE source_ingestion_runs
                SET status = 'FAILED', completed_at = CURRENT_TIMESTAMP,
                    error_message = 'Ingestion interrotta recuperata automaticamente'
                WHERE feed_id = :feed AND season_id = :season
                  AND status = 'STARTED'
                  AND started_at < CURRENT_TIMESTAMP - (:staleSeconds * INTERVAL '1 second')
                """)
                .param("feed", feed.getId())
                .param("season", seasonId)
                .param("staleSeconds", config.getStaleThreshold().toSeconds())
                .update();
        try {
            return jdbc.sql("""
                    INSERT INTO source_ingestion_runs (
                        feed_id, feed_type, source, season_id, sanitized_url, status
                    )
                    VALUES (:id, :type, :source, :season, :url, 'STARTED')
                    RETURNING id
                    """)
                    .param("id", feed.getId())
                    .param("type", type(feed).name())
                    .param("source", feed.getSource().trim().toUpperCase(Locale.ROOT))
                    .param("season", seasonId)
                    .param("url", RemoteCsvClient.sanitized(remote.validate(feed.getUrl())))
                    .query(Long.class)
                    .single();
        } catch (DataIntegrityViolationException exception) {
            throw new ConflictException("Ingestion feed già in esecuzione");
        }
    }

    private void finish(
            long id,
            String status,
            Integer httpStatus,
            Long payloadBytes,
            String checksum,
            Long batchId,
            String error

                    error_message = :error,

                """)
                .param("status", status)
                .param("http", httpStatus)
                .param("bytes", payloadBytes)
                .param("sha", checksum)
                .param("batch", batchId)
                .param("error", error)
                .param("id", id)
                .update();
    }

        String table = switch (feedType) {

            case STATISTICS -> "player_statistics_import_batches";

        jdbc.sql("UPDATE " + table + " SET source_observed_at = CURRENT_TIMESTAMP WHERE id = :id")
                .param("id", batchId)
                .update();
    }

package it.fantacalciointelligence.backend.reconciliation;

import java.sql.Types;
import java.time.OffsetDateTime;
import java.util.LinkedHashSet;

import java.util.Map;

import java.util.Set;

import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import it.fantacalciointelligence.backend.common.ConflictException;
import it.fantacalciointelligence.backend.common.ResourceNotFoundException;
import it.fantacalciointelligence.backend.reconciliation.identity.PlayerIdentityResolutionService;
import it.fantacalciointelligence.backend.reconciliation.identity.ResolvedPlayerIdentity;

@Service
public class PlayerReconciliationService {

    private static final String FANTACALCIO = "FANTACALCIO";
    private static final String GAZZETTA = "GAZZETTA";
    private static final String FANTAMARKT = "FANTAMARKT";

    private final JdbcClient jdbcClient;
    private final PlayerIdentityResolutionService identityResolutionService;

    public PlayerReconciliationService(
            JdbcClient jdbcClient,
            PlayerIdentityResolutionService identityResolutionService
    ) {
        this.jdbcClient = jdbcClient;
        this.identityResolutionService = identityResolutionService;
    }

    @Transactional
    public ReconciliationResponse reconcile(
            String seasonCode
    ) {

        Long seasonId =
                findSeasonId(seasonCode);

        BatchSnapshot fantacalcioBatch =
                latestBatch(
                        seasonId,
                        FANTACALCIO
                );

        BatchSnapshot gazzettaBatch =
                latestBatch(
                        seasonId,
                        GAZZETTA
                );

        Optional<BatchSnapshot> fantamarktBatch =
                latestOptionalProductionBatch(
                        seasonId,
                        FANTAMARKT
                );

        markReconciliationStarted(
                fantacalcioBatch.id()
        );

        markReconciliationStarted(
                gazzettaBatch.id()
        );

        fantamarktBatch.ifPresent(batch ->
                markReconciliationStarted(
                        batch.id()
                )
        );

        List<ResolvedPlayerIdentity> identities =
                identityResolutionService.resolve(
                        fantacalcioBatch.id(),
                        gazzettaBatch.id(),
                        fantamarktBatch
                                .map(BatchSnapshot::id)
                                .orElse(null)
                );

        /*
         * Fantacalcio is the authoritative operational catalog
         * during the transfer market.
         *
         * Secondary sources enrich and confirm identity/data,
         * but they do not decide whether a current Fantacalcio
         * player can exist in the operational catalog.
         */
        List<ResolvedPlayerIdentity> catalogIdentities =
                identities.stream()
                        .filter(identity ->
                                isNotBlank(
                                        identity
                                                .fantacalcioName()
                                )
                        )
                        .toList();

        /*
         * Kept for API backward compatibility.
         *
         * Single-source Fantacalcio players are no longer
         * excluded from the operational catalog. Their lack of
         * secondary confirmation remains visible through the
         * reconciliation preview / enrichment workflow.
         */
        int excludedSingleSourcePlayers = 0;

        int matchedPlayers = 0;
        int matchedWithThreeSources = 0;
        int matchedWithTwoSources = 0;
        int createdPlayers = 0;
        int updatedPlayers = 0;

        int fantacalcioReconciledRows = 0;
        int gazzettaReconciledRows = 0;
        int fantamarktReconciledRows = 0;

        /*
         * Rebuild the official current catalog transactionally.
         */
        jdbcClient.sql("""
                DELETE FROM player_reconciliation_results
                WHERE season_id = :seasonId
                """)
                .param("seasonId", seasonId)
                .update();

        /*
         * Operational lifecycle follows the latest Fantacalcio
         * listone snapshot.
         *
         * A player that was ACTIVE but is no longer present in
         * the latest snapshot becomes INACTIVE.
         *
         * No player/history/source mapping is deleted.
         *
         * If the player appears again in a later Fantacalcio
         * snapshot, saveRegistration(...) below reactivates him.
         */
        deactivatePlayersMissingFromLatestFantacalcio(
                seasonId,
                fantacalcioBatch.id()
        );

        for (ResolvedPlayerIdentity identity
                : catalogIdentities) {

            /*
             * Fantacalcio is authoritative by default.
             *
             * A durable, explicitly verified EXCLUDED review is
             * the controlled exception: it means that this player
             * has already been manually verified as not belonging
             * to the operational Serie A catalog even if the
             * Fantacalcio snapshot is temporarily stale.
             */
            if (isExplicitlyExcluded(
                    seasonId,
                    identity.fantacalcioExternalId()
            )) {

                findMappedPlayerId(
                        FANTACALCIO,
                        identity.fantacalcioExternalId()
                ).ifPresent(playerId ->
                        deactivateRegistration(
                                playerId,
                                seasonId
                        )
                );

                continue;
            }

            String displayName =
                    firstNonBlank(
                            identity.fantacalcioName(),
                            identity.gazzettaName(),
                            identity.fantamarktName()
                    );

            Optional<Long> existingPlayerId =
                    findExistingPlayerId(
                            identity
                    );

            Long playerId;

            if (existingPlayerId.isPresent()) {

                playerId =
                        existingPlayerId.get();

                updatePlayer(
                        playerId,
                        displayName,
                        identity.normalizedName()
                );

                updatedPlayers++;

            } else {

                playerId =
                        createPlayer(
                                displayName,
                                identity.normalizedName()
                        );

                createdPlayers++;
            }

            /*
             * Source mappings are identity information.
             * They are saved independently from club resolution.
             */
            saveSourceIfPresent(
                    playerId,
                    seasonId,
                    FANTACALCIO,
                    identity.fantacalcioExternalId(),
                    identity.fantacalcioName(),
                    identity.fantacalcioRole()
            );

            saveMarketValue(
                    playerId,
                    seasonId,
                    FANTACALCIO,
                    identity.fantacalcioQuotation(),
                    identity.fantacalcioFmv()
            );

            saveSourceIfPresent(
                    playerId,
                    seasonId,
                    GAZZETTA,
                    identity.gazzettaExternalId(),
                    identity.gazzettaName(),
                    identity.gazzettaRole()
            );

            saveSourceIfPresent(
                    playerId,
                    seasonId,
                    FANTAMARKT,
                    identity.fantamarktExternalId(),
                    identity.fantamarktName(),
                    identity.fantamarktRole()
            );

            /*
             * Fantacalcio marks players that are still visible in
             * the listone but no longer play in Serie A with the
             * semantic out-of-game marker.
             *
             * They remain historically stored with their mappings,
             * roles and market data, but must not be part of the
             * current operational catalog.
             *
             * TransferDetectionService separately creates a
             * DETECTED exit claim. Official evidence is still
             * required before assigning a definitive transfer
             * lifecycle such as RELEASED.
             */
            if (isFantacalcioOutOfGame(
                    fantacalcioBatch.id(),
                    identity.fantacalcioExternalId()
            )) {

                deactivateRegistration(
                        playerId,
                        seasonId
                );

                continue;
            }

            ClubResolution clubResolution =
                    resolveCurrentClub(
                            playerId,
                            seasonId,
                            identity,
                            fantacalcioBatch
                                    .sourceObservedAt()
                    );

            /*
             * An official and sufficiently recent exit from
             * Serie A wins over an older Fantacalcio snapshot.
             *
             * The player remains stored historically but is no
             * longer included in the current official catalog.
             */
            if (clubResolution.leftSerieA()) {

                preserveReleasedRegistration(
                        playerId,
                        seasonId
                );

                continue;
            }

            saveRegistration(
                    playerId,
                    clubResolution.clubId(),
                    seasonId
            );

            saveReconciliationResult(
                    seasonId,
                    playerId,
                    identity,
                    clubResolution
            );

            matchedPlayers++;

            if (identity.sourcesCount() == 3) {
                matchedWithThreeSources++;
            } else if (identity.sourcesCount() == 2) {
                matchedWithTwoSources++;
            }

            fantacalcioReconciledRows++;

            if (isNotBlank(
                    identity.gazzettaName()
            )) {
                gazzettaReconciledRows++;
            }

            if (isNotBlank(
                    identity.fantamarktName()
            )) {
                fantamarktReconciledRows++;
            }
        }

        markReconciliationCompleted(
                fantacalcioBatch.id(),
                fantacalcioReconciledRows
        );

        markReconciliationCompleted(
                gazzettaBatch.id(),
                gazzettaReconciledRows
        );

        if (fantamarktBatch.isPresent()) {
            markReconciliationCompleted(
                    fantamarktBatch.get().id(),
                    fantamarktReconciledRows
            );
        }

        return new ReconciliationResponse(
                seasonCode,
                fantacalcioBatch.id(),
                gazzettaBatch.id(),
                fantamarktBatch
                        .map(BatchSnapshot::id)
                        .orElse(0L),
                matchedPlayers,
                matchedWithThreeSources,
                matchedWithTwoSources,
                excludedSingleSourcePlayers,
                createdPlayers,
                updatedPlayers,
                "COMPLETED"
        );
    }

    private boolean isFantacalcioOutOfGame(
            Long fantacalcioBatchId,
            String fantacalcioExternalId
    ) {

        if (!isNotBlank(fantacalcioExternalId)) {
            return false;
        }

        Boolean outOfGame =
                jdbcClient.sql("""
                        SELECT EXISTS (
                            SELECT 1
                            FROM player_import_rows
                            WHERE batch_id = :batchId
                              AND external_id =
                                  :fantacalcioExternalId
                              AND valid = TRUE
                              AND LOWER(
                                    COALESCE(
                                        raw_data ->> 'out_of_game',
                                        'false'
                                    )
                                  ) = 'true'
                        )
                        """)
                        .param(
                                "batchId",
                                fantacalcioBatchId
                        )
                        .param(
                                "fantacalcioExternalId",
                                fantacalcioExternalId
                        )
                        .query(Boolean.class)
                        .single();

        return Boolean.TRUE.equals(outOfGame);
    }
    private boolean isExplicitlyExcluded(
            Long seasonId,
            String fantacalcioExternalId
    ) {

        if (!isNotBlank(fantacalcioExternalId)) {
            return false;
        }

        Integer count =
                jdbcClient.sql("""
                        SELECT COUNT(*)
                        FROM reconciliation_review_items
                        WHERE season_id = :seasonId
                          AND fantacalcio_external_id =
                              :fantacalcioExternalId
                          AND review_status = 'EXCLUDED'
                        """)
                        .param(
                                "seasonId",
                                seasonId
                        )
                        .param(
                                "fantacalcioExternalId",
                                fantacalcioExternalId
                        )
                        .query(Integer.class)
                        .single();

        return count > 0;
    }

    private void deactivateRegistration(
            Long playerId,
            Long seasonId
    ) {

        jdbcClient.sql("""
                UPDATE player_registrations

    private ClubResolution resolveCurrentClub(
            Long playerId,
            Long seasonId,
            ResolvedPlayerIdentity identity,
            OffsetDateTime fantacalcioObservedAt
    ) {

        Optional<OfficialTransferClaim> officialClaim =
                findLatestOfficialTransferClaim(
                        playerId,
                        seasonId
                );

        if (officialClaim.isPresent()) {

            OfficialTransferClaim claim =
                    officialClaim.get();

            OffsetDateTime officialTimestamp =
                    firstNonNull(
                            claim.sourcePublishedAt(),
                            claim.verifiedAt(),
                            claim.observedAt()
                    );

            /*
             * If we have trustworthy temporal provenance for
             * Fantacalcio and that snapshot is strictly newer,
             * Fantacalcio becomes the current source again.
             *
             * For legacy imports sourceObservedAt is NULL:
             * in that case an official verified event wins.
             */
            boolean fantacalcioIsNewer =
                    fantacalcioObservedAt != null
                            && officialTimestamp != null
                            && officialTimestamp.isBefore(
                                    fantacalcioObservedAt
                            );

            if (!fantacalcioIsNewer) {

                if (claim.toClubId() == null) {

                    return new ClubResolution(
                            null,
                            null,
                            null,
                            claim.source(),
                            claim.verifiedAt(),
                            true
                    );
                }

                return new ClubResolution(
                        claim.toClubId(),
                        claim.toClubName(),
                        normalizeClub(
                                claim.toClubName()
                        ),
                        claim.source(),
                        claim.verifiedAt(),
                        false
                );
            }
        }

        Long fantacalcioClubId =
                findClubId(
                        identity.fantacalcioClub()
                );

        return new ClubResolution(
                fantacalcioClubId,
                identity.fantacalcioClub(),
                firstNonBlank(
                        identity
                                .fantacalcioNormalizedClub(),
                        normalizeClub(
                                identity.fantacalcioClub()
                        )
                ),
                FANTACALCIO,
                null,
                false
        );
    }

    private Optional<Long> findExistingPlayerId(
            ResolvedPlayerIdentity identity
    ) {

        Set<Long> mappedPlayerIds =
                new LinkedHashSet<>();

        findMappedPlayerId(
                FANTACALCIO,
                identity.fantacalcioExternalId()
        ).ifPresent(mappedPlayerIds::add);

        findMappedPlayerId(
                GAZZETTA,
                identity.gazzettaExternalId()
        ).ifPresent(mappedPlayerIds::add);

        findMappedPlayerId(
                FANTAMARKT,
                identity.fantamarktExternalId()
        ).ifPresent(mappedPlayerIds::add);

        if (mappedPlayerIds.size() > 1) {
            throw new ConflictException(
                    "Le fonti risultano associate "
                            + "a giocatori diversi: "
                            + identity.normalizedName()
            );
        }

        if (mappedPlayerIds.size() == 1) {
            return mappedPlayerIds
                    .stream()
                    .findFirst();
        }

        List<Long> sameNamePlayerIds =
                jdbcClient.sql("""
                        SELECT id
                        FROM players
                        WHERE normalized_name =
                              :normalizedName
                        ORDER BY id
                        """)
                        .param(
                                "normalizedName",
                                identity.normalizedName()
                        )
                        .query(Long.class)
                        .list();

        if (sameNamePlayerIds.isEmpty()) {
            return Optional.empty();
        }

        if (sameNamePlayerIds.size() == 1) {
            return Optional.of(
                    sameNamePlayerIds.getFirst()
            );
        }

        /*
         * Club is deliberately NOT used here.
         * Identity and current registration are independent.
         */
        throw new ConflictException(
                "IdentitÃƒÂ  interna ambigua per il giocatore: "
                        + identity.normalizedName()
        );
    }

    private void saveSourceMapping(
            Long playerId,
            String source,
            String externalId,
            String externalName
    ) {

        Optional<Map<String, Object>> existingMapping;

        if (isNotBlank(externalId)) {

            existingMapping =
                    findMappingByExternalId(
                            source,
                            externalId
                    );

        } else {

            existingMapping =
                    findUniqueMappingByExternalName(
                            source,
                            externalName
                    );
        }

        if (existingMapping.isPresent()) {

            Long existingPlayerId =
                    numberValue(
                            existingMapping.get(),
                            "player_id"
                    );

            if (!existingPlayerId.equals(
                    playerId
            )) {

                throw new ConflictException(
                        "Il mapping "
                                + source
                                + " ÃƒÂ¨ giÃƒÂ  associato "
                                + "a un altro giocatore: "
                                + externalName
                );
            }

            Long mappingId =
                    numberValue(
                            existingMapping.get(),
                            "id"
                    );

            jdbcClient.sql("""
                    UPDATE player_source_mappings
                    SET external_id = :externalId,
                        external_name = :externalName,
                        verified = TRUE,
                        updated_at = CURRENT_TIMESTAMP
                    WHERE id = :mappingId
                    """)
                    .param(
                            "externalId",
                            externalId,

                    )
                    .param(
                            "externalName",
                            externalName
                    )
                    .param(
                            "mappingId",
                            mappingId
                    )
                    .update();

            return;
        }

        jdbcClient.sql("""
                INSERT INTO player_source_mappings (
                    player_id,
                    source,
                    external_id,
                    external_name,
                    verified
                )
                VALUES (
                    :playerId,
                    :source,
                    :externalId,

                    TRUE
                )

                .param("playerId", playerId)
                .param("source", source)
                .param(

    }

    private String firstNonBlank(
            String... values
    ) {

        for (String value : values) {
            if (isNotBlank(value)) {
                return value.trim();
            }
        }

        throw new ConflictException(
                "Nessun valore valido disponibile "
                        + "durante la riconciliazione"
        );
    }

            String club
    ) {

        if (!isNotBlank(club)) {

    @SafeVarargs
    private final <T> T firstNonNull(
            T... values
    ) {

        for (T value : values) {
            if (value != null) {
                return value;
            }
        }

        return null;
    }

    private boolean isNotBlank(
            String value
    ) {
        return value != null
                && !value.isBlank();
    }

    private Long numberValue(
            Map<String, Object> row,

    ) {

    private Long nullableLong(
            java.sql.ResultSet resultSet,
            String column
    ) throws java.sql.SQLException {

        long value =
                resultSet.getLong(column);

        return resultSet.wasNull()
                ? null
                : value;

        List<CatalogQualityIssue> issues = new ArrayList<>(qualityReport.issues());
        addReconciledNotOperationalIssues(
                season, fantacalcioBatchId, preview, qualityReport, issues);
        freshness.blockers().forEach(message -> issues.add(issue(
                "ERROR", "DATA_FRESHNESS_BLOCKER", message, null, null, null, null, Map.of())));
        freshness.warnings().forEach(message -> issues.add(issue(
                "WARNING", "DATA_FRESHNESS_WARNING", message, null, null, null, null, Map.of())));

        if (preview.totalFantacalcioPlayers() == 0) {
            issues.add(issue("ERROR", "EMPTY_FANTACALCIO_BATCH",
                    "Il batch Fantacalcio non contiene giocatori", null, null, null, null, Map.of()));
        }
        if (preview.unresolvedPlayers() > 0) {
            issues.add(issue("WARNING", "UNRESOLVED_PREVIEW_IDENTITIES",
                    preview.unresolvedPlayers() + " giocatori Fantacalcio non sono confermati nella preview",
                    null, null, null, null, Map.of("count", preview.unresolvedPlayers())));
        }

        int pendingTransfers = countPendingTransfers(season);
        int officialExits = countOfficialExits(season);
        int missingFromFantacalcio = countMissingFromFantacalcio(season, fantacalcioBatchId);
        if (pendingTransfers > 0) {
            issues.add(issue("WARNING", "PENDING_TRANSFER_CLAIMS",
                    pendingTransfers + " trasferimenti richiedono verifica", null, null, null, null,
                    Map.of("count", pendingTransfers)));
        }
        if (missingFromFantacalcio > 0) {
            issues.add(issue("WARNING", "MISSING_FROM_LATEST_FANTACALCIO_SNAPSHOT",
                    missingFromFantacalcio + " giocatori operativi non compaiono nell'ultimo snapshot Fantacalcio",
                    null, null, null, null, Map.of("count", missingFromFantacalcio)));
        }

        List<CatalogQualityIssue> immutableIssues = List.copyOf(issues);
        int errorCount = countSeverity(immutableIssues, "ERROR");
        int warningCount = countSeverity(immutableIssues, "WARNING");
        String status = errorCount > 0 ? "BLOCKED" : "READY";
        List<String> blockers = messages(immutableIssues, "ERROR");
        List<String> warnings = messages(immutableIssues, "WARNING");

        return new CatalogReadinessResponse(
                season, fantacalcioBatchId, gazzettaBatchId, fantamarktBatchId,
                preview.totalFantacalcioPlayers(), preview.confirmedPlayers(), preview.unresolvedPlayers(),
                qualityReport.totalOperationalPlayers(), qualityReport.playersWithoutFantacalcioRole(),
                qualityReport.playersWithoutActiveRegistration(), pendingTransfers, officialExits,
                missingFromFantacalcio, calculateQuality(preview.confirmedPlayers(), preview.totalFantacalcioPlayers()),
                freshness, status, blockers, warnings, status, qualityReport.totalOperationalPlayers(),
                qualityReport.clubSummaries().size(), qualityReport.playersByRole(), errorCount, warningCount,
                immutableIssues, qualityReport.clubSummaries());
    }

    QualityReport evaluateCatalogQuality(String season) {
        requireSeason(season);
        List<CatalogQualityIssue> issues = new ArrayList<>();
        List<OperationalPlayer> players = loadOperationalPlayers(season);

        for (OperationalPlayer player : players) {
            validatePlayer(player, issues);
        }
        addDuplicateIssues(players, issues);
        addOpenReviewIssues(season, issues);
        addExcludedCatalogIssues(season, issues);
        addVerifiedLinkIssues(season, issues);

        List<CatalogClubSummary> clubs = loadClubSummaries(season);
        addClubIssues(clubs, issues);
        CatalogRoleSummary roles = summarizeRoles(players);

        return new QualityReport(
                players.size(),
                (int) players.stream().filter(player -> player.role() == null || player.role().isBlank()).count(),
                (int) players.stream().filter(player -> player.registrationId() == null).count(),
                roles,
                players.stream().map(OperationalPlayer::fantacalcioExternalId)
                        .filter(id -> id != null && !id.isBlank())
                        .collect(java.util.stream.Collectors.toUnmodifiableSet()),
                List.copyOf(clubs), List.copyOf(issues));
    }

    private void validatePlayer(OperationalPlayer player, List<CatalogQualityIssue> issues) {
        if (player.playerId() == null) {
            issues.add(playerIssue("MISSING_PLAYER_ID", "Giocatore operativo senza ID", player));
        }
        if (player.name() == null || player.name().isBlank()) {
            issues.add(playerIssue("MISSING_PLAYER_NAME", "Giocatore operativo senza nome", player));
        }
        if (player.registrationId() == null || player.clubId() == null || player.club() == null) {
            issues.add(playerIssue("MISSING_CLUB", "Giocatore operativo senza club per la stagione", player));
        } else if (!player.clubActive()) {
            issues.add(playerIssue("INVALID_CLUB", "Giocatore associato a un club non attivo", player));
        }
        if (player.role() == null || player.role().isBlank()) {
            issues.add(playerIssue("MISSING_FANTACALCIO_ROLE", "Ruolo Fantacalcio mancante", player));
        } else if (!VALID_ROLES.contains(player.role())) {
            issues.add(playerIssue("INVALID_FANTACALCIO_ROLE", "Ruolo Fantacalcio non valido: " + player.role(), player));
        }
        if (player.fantacalcioExternalId() == null || player.fantacalcioExternalId().isBlank()) {
            issues.add(playerIssue("MISSING_FANTACALCIO_SOURCE", "Fonte identitaria Fantacalcio mancante", player));
        }
        /*
         * Fantacalcio is sufficient for operational inclusion.
         *
         * Secondary identity sources are enrichment/confirmation.
         * A Fantacalcio-only player is therefore valid while
         * enrichment is pending.
         */

        /*
         * sources_count belongs to the latest reconciliation
         * snapshot, while player_source_mappings is persistent
         * identity history.
         *
         * Current snapshot sources may legitimately be fewer than
         * persistent verified mappings.
         *
         * Invalid cases are:
         * - no source at all;
         * - reconciliation claiming more sources than the verified
         *   persistent identity mappings.
         */
        if (player.reconciledSources() < 1
                || player.reconciledSources() > player.identitySources()) {

            issues.add(issue(
                    "ERROR",
                    "RECONCILIATION_SOURCE_MISMATCH",
                    "Conteggio fonti della reconciliation incoerente con i mapping verificati",
                    player.playerId(),
                    player.fantacalcioExternalId(),
                    player.name(),
                    player.club(),
                    Map.of(
                            "reconciliationSources",
                            player.reconciledSources(),
                            "verifiedIdentitySources",
                            player.identitySources()
                    )
            ));
        }
    }

    private void addDuplicateIssues(List<OperationalPlayer> players, List<CatalogQualityIssue> issues) {
        Map<String, List<OperationalPlayer>> byFantacalcioId = new LinkedHashMap<>();
        players.stream()
                .filter(player -> player.fantacalcioExternalId() != null && !player.fantacalcioExternalId().isBlank())
                .forEach(player -> byFantacalcioId.computeIfAbsent(
                        player.fantacalcioExternalId(), ignored -> new ArrayList<>()).add(player));
        byFantacalcioId.values().stream().filter(group -> group.size() > 1).forEach(group ->
                group.forEach(player -> issues.add(issue("ERROR", "DUPLICATE_OPERATIONAL_PLAYER",
                        "IdentitÃ  Fantacalcio duplicata nel catalogo operativo", player.playerId(),
                        player.fantacalcioExternalId(), player.name(), player.club(),
                        Map.of("duplicateCount", group.size())))));
    }

    private void addOpenReviewIssues(String season, List<CatalogQualityIssue> issues) {
        jdbcClient.sql("""
                SELECT review.id, review.fantacalcio_external_id, review.fantacalcio_name,
                       review.fantacalcio_club
                FROM reconciliation_review_items review
                JOIN seasons season ON season.id = review.season_id
                WHERE season.code = :season AND review.review_status = 'OPEN'
                ORDER BY review.id
                """).param("season", season).query((rs, rowNum) -> issue(
                        "ERROR", "OPEN_RECONCILIATION_REVIEW", "Review di reconciliation ancora aperta",
                        null, rs.getString("fantacalcio_external_id"), rs.getString("fantacalcio_name"),
                        rs.getString("fantacalcio_club"), Map.of("reviewId", rs.getLong("id"))))
                .list().forEach(issues::add);
    }

    private void addExcludedCatalogIssues(String season, List<CatalogQualityIssue> issues) {
        jdbcClient.sql("""
                SELECT review.id, review.fantacalcio_external_id, review.fantacalcio_name,
                       review.fantacalcio_club, mapping.player_id
                FROM reconciliation_review_items review
                JOIN seasons season ON season.id = review.season_id
                JOIN player_source_mappings mapping
                  ON mapping.source = 'FANTACALCIO'
                 AND mapping.external_id = review.fantacalcio_external_id
                JOIN player_reconciliation_results result
                  ON result.player_id = mapping.player_id AND result.season_id = review.season_id
                WHERE season.code = :season AND review.review_status = 'EXCLUDED'
                ORDER BY review.id
                """).param("season", season).query((rs, rowNum) -> issue(
                        "ERROR", "EXCLUDED_PLAYER_IN_CATALOG", "Giocatore EXCLUDED presente nel catalogo operativo",
                        rs.getLong("player_id"), rs.getString("fantacalcio_external_id"),
                        rs.getString("fantacalcio_name"), rs.getString("fantacalcio_club"),
                        Map.of("reviewId", rs.getLong("id"))))
                .list().forEach(issues::add);
    }

    private void addVerifiedLinkIssues(String season, List<CatalogQualityIssue> issues) {
        jdbcClient.sql("""
                SELECT fc.player_id, fc.external_id AS fc_id, player.display_name,
                       link.secondary_source, link.secondary_external_id,
                       secondary.player_id AS secondary_player_id
                FROM verified_source_identity_links link
                JOIN player_source_mappings fc
                  ON fc.source = 'FANTACALCIO' AND fc.external_id = link.fantacalcio_external_id
                JOIN player_reconciliation_results result ON result.player_id = fc.player_id
                JOIN seasons season ON season.id = result.season_id
                JOIN players player ON player.id = fc.player_id
                LEFT JOIN player_source_mappings secondary
                  ON secondary.source = link.secondary_source
                 AND secondary.external_id = link.secondary_external_id
                WHERE season.code = :season AND link.verified = TRUE
                  AND (secondary.player_id IS NULL OR secondary.player_id <> fc.player_id)
                """).param("season", season).query((rs, rowNum) -> issue(
                        "ERROR", "VERIFIED_MAPPING_INCONSISTENCY",
                        "Un mapping verificato non Ã¨ rispettato dal catalogo operativo",
                        rs.getLong("player_id"), rs.getString("fc_id"), rs.getString("display_name"), null,
                        Map.of("secondarySource", rs.getString("secondary_source"),
                                "secondaryExternalId", rs.getString("secondary_external_id"))))
                .list().forEach(issues::add);
    }

    private void addReconciledNotOperationalIssues(
            String season,
            long fantacalcioBatchId,
            ReconciliationPreviewResponse preview,
            QualityReport qualityReport,
            List<CatalogQualityIssue> issues
    ) {
        preview.players().stream()
                .filter(player -> "CONFIRMED".equals(player.status()))
                .filter(player -> !qualityReport.operationalFantacalcioIds()
                        .contains(player.fantacalcioExternalId()))
                .filter(player -> !isExcluded(season, player.fantacalcioExternalId()))
                .filter(player -> !isCurrentOfficialExit(
                        season, fantacalcioBatchId, player.fantacalcioExternalId()))
                .map(player -> reconciledNotOperationalIssue(season, player))
                .forEach(issues::add);
    }

    private boolean isExcluded(String season, String fantacalcioExternalId) {
        return jdbcClient.sql("""
                SELECT EXISTS (
                    SELECT 1 FROM reconciliation_review_items review
                    JOIN seasons season ON season.id = review.season_id
                    WHERE season.code = :season
                      AND review.fantacalcio_external_id = :externalId
                      AND review.review_status = 'EXCLUDED')
                """).param("season", season).param("externalId", fantacalcioExternalId)
                .query(Boolean.class).single();
    }

    private boolean isCurrentOfficialExit(
            String season,
            long fantacalcioBatchId,
            String fantacalcioExternalId
    ) {
        return jdbcClient.sql("""
                WITH latest_claim AS (
                    SELECT claim.to_club_id,
                           COALESCE(claim.source_published_at, claim.verified_at, claim.observed_at)
                               AS official_at
                    FROM player_source_mappings mapping
                    JOIN player_transfer_claims claim ON claim.player_id = mapping.player_id
                    JOIN seasons season ON season.id = claim.season_id
                    WHERE mapping.source = 'FANTACALCIO'
                      AND mapping.external_id = :externalId
                      AND season.code = :season
                      AND claim.verification_status = 'OFFICIAL'
                      AND claim.source IN ('FIGC', 'LEGA_SERIE_A', 'OFFICIAL_CLUB')
                    ORDER BY COALESCE(
                            claim.source_published_at, claim.verified_at, claim.observed_at) DESC,
                             claim.id DESC
                    LIMIT 1)
                SELECT EXISTS (
                    SELECT 1 FROM latest_claim claim
                    JOIN import_batches batch ON batch.id = :batchId
                    WHERE claim.to_club_id IS NULL
                      AND NOT (batch.source_observed_at IS NOT NULL
                               AND claim.official_at IS NOT NULL
                               AND claim.official_at < batch.source_observed_at))
                """).param("externalId", fantacalcioExternalId).param("season", season)
                .param("batchId", fantacalcioBatchId).query(Boolean.class).single();
    }

    private CatalogQualityIssue reconciledNotOperationalIssue(
            String season,
            ReconciliationPreviewPlayer previewPlayer
    ) {
        ReconciledPlayerState state = loadReconciledPlayerState(
                season, previewPlayer.fantacalcioExternalId());
        Map<String, Object> details = new LinkedHashMap<>();
        details.put("previewSourcesCount", previewPlayer.sourcesCount());
        putIfNotNull(details, "identityResolutionMethod", previewPlayer.identityResolutionMethod());
        putIfNotNull(details, "playerActive", state.active());
        putIfNotNull(details, "registrationStatus", state.registrationStatus());
        putIfNotNull(details, "fantacalcioRole", state.role());
        putIfNotNull(details, "reviewStatus", state.reviewStatus());
        putIfNotNull(details, "sourceMappings", state.mappings());

        return issue("ERROR", "RECONCILED_PLAYER_NOT_OPERATIONAL",
                "IdentitÃ  confermata dalla reconciliation corrente ma assente dal catalogo operativo",
                state.playerId(), previewPlayer.fantacalcioExternalId(),
                state.playerName() != null ? state.playerName() : previewPlayer.fantacalcioName(),
                state.club() != null ? state.club() : previewPlayer.fantacalcioClub(), details);
    }

    private ReconciledPlayerState loadReconciledPlayerState(
            String season,
            String fantacalcioExternalId
    ) {
        return jdbcClient.sql("""
                SELECT player.id AS player_id, player.display_name, player.active,
                       registration.registration_status, club.name AS club_name,
                       role.fantasy_role, review.review_status,
                       STRING_AGG(DISTINCT source_mapping.source || ':' ||
                           COALESCE(source_mapping.external_id, '<null>'), ', ') AS mappings
                FROM player_source_mappings mapping
                JOIN players player ON player.id = mapping.player_id
                JOIN seasons season ON season.code = :season
                LEFT JOIN player_registrations registration
                  ON registration.player_id = player.id AND registration.season_id = season.id
                LEFT JOIN clubs club ON club.id = registration.club_id
                LEFT JOIN player_roles role
                  ON role.player_id = player.id AND role.season_id = season.id
                 AND role.source = 'FANTACALCIO'
                LEFT JOIN reconciliation_review_items review
                  ON review.season_id = season.id
                 AND review.fantacalcio_external_id = mapping.external_id
                LEFT JOIN player_source_mappings source_mapping ON source_mapping.player_id = player.id
                WHERE mapping.source = 'FANTACALCIO' AND mapping.external_id = :externalId
                GROUP BY player.id, player.display_name, player.active,
                         registration.registration_status, club.name,
                         role.fantasy_role, review.review_status
                """).param("season", season).param("externalId", fantacalcioExternalId)
                .query((rs, rowNum) -> new ReconciledPlayerState(
                        rs.getObject("player_id", Long.class), rs.getString("display_name"),
                        rs.getObject("active", Boolean.class), rs.getString("registration_status"),
                        rs.getString("club_name"), rs.getString("fantasy_role"),
                        rs.getString("review_status"), rs.getString("mappings")))
                .optional().orElse(new ReconciledPlayerState(
                        null, null, null, null, null, null, null, null));
    }

    private void putIfNotNull(Map<String, Object> details, String key, Object value) {
        if (value != null) {
            details.put(key, value);
        }
    }

    private void addClubIssues(List<CatalogClubSummary> clubs, List<CatalogQualityIssue> issues) {
        for (CatalogClubSummary club : clubs) {
            if (club.totalPlayers() == 0) {
                issues.add(issue("ERROR", "EMPTY_CLUB_ROSTER", "Club della competizione senza giocatori operativi",
                        null, null, null, club.club(), Map.of("clubId", club.clubId())));
                continue;
            }
            if (club.totalPlayers() < SUSPICIOUS_MIN_TOTAL || club.P() < SUSPICIOUS_MIN_GOALKEEPERS) {
                issues.add(issue("WARNING", "SUSPICIOUS_CLUB_ROSTER", "Rosa insolitamente ridotta",
                        null, null, null, club.club(), Map.of("totalPlayers", club.totalPlayers(), "goalkeepers", club.P())));
            }
            int largestRole = Math.max(Math.max(club.P(), club.D()), Math.max(club.C(), club.A()));
            if (largestRole * 100 > club.totalPlayers() * SUSPICIOUS_MAX_ROLE_SHARE_PERCENT) {
                issues.add(issue("WARNING", "SUSPICIOUS_ROLE_DISTRIBUTION", "Distribuzione ruoli insolitamente sbilanciata",
                        null, null, null, club.club(), Map.of("largestRoleSharePercent",
                                largestRole * 100 / club.totalPlayers())));
            }
        }
    }

    private List<OperationalPlayer> loadOperationalPlayers(String season) {
        return jdbcClient.sql("""
                SELECT result.player_id, player.display_name, registration.id AS registration_id,
                       registration.club_id, club.name AS club_name,
                       (season_club.club_id IS NOT NULL) AS club_active,
                       role.fantasy_role,
                       MAX(mapping.external_id) FILTER (WHERE mapping.source = 'FANTACALCIO' AND mapping.verified = TRUE)
                           AS fantacalcio_external_id,
                       COUNT(DISTINCT mapping.source) FILTER (
                           WHERE mapping.verified = TRUE AND mapping.source IN ('FANTACALCIO', 'GAZZETTA', 'FANTAMARKT'))
                           AS identity_sources,
                       result.sources_count
                FROM player_reconciliation_results result
                JOIN seasons season ON season.id = result.season_id
                LEFT JOIN players player ON player.id = result.player_id
                LEFT JOIN player_registrations registration
                  ON registration.player_id = result.player_id AND registration.season_id = result.season_id
                 AND registration.registration_status = 'ACTIVE'
                LEFT JOIN clubs club ON club.id = registration.club_id
                LEFT JOIN season_clubs season_club
                  ON season_club.season_id = result.season_id AND season_club.club_id = registration.club_id
                LEFT JOIN player_roles role
                  ON role.player_id = result.player_id AND role.season_id = result.season_id
                 AND role.source = 'FANTACALCIO'
                LEFT JOIN player_source_mappings mapping ON mapping.player_id = result.player_id
                WHERE season.code = :season AND COALESCE(player.active, FALSE) = TRUE
                GROUP BY result.player_id, player.display_name, registration.id, registration.club_id,
                         club.name, season_club.club_id, role.fantasy_role, result.sources_count
                ORDER BY result.player_id
                """).param("season", season).query((rs, rowNum) -> new OperationalPlayer(
                        rs.getObject("player_id", Long.class), rs.getString("display_name"),
                        rs.getObject("registration_id", Long.class), rs.getObject("club_id", Long.class),
                        rs.getString("club_name"), rs.getBoolean("club_active"), rs.getString("fantasy_role"),
                        rs.getString("fantacalcio_external_id"), rs.getInt("identity_sources"),
                        rs.getInt("sources_count"))).list();
    }

    private List<CatalogClubSummary> loadClubSummaries(String season) {
        return jdbcClient.sql("""
                SELECT club.id, club.name,
                       COUNT(player.id) AS total_players,
                       COUNT(player.id) FILTER (WHERE role.fantasy_role = 'P') AS p,
                       COUNT(player.id) FILTER (WHERE role.fantasy_role = 'D') AS d,
                       COUNT(player.id) FILTER (WHERE role.fantasy_role = 'C') AS c,
                       COUNT(player.id) FILTER (WHERE role.fantasy_role = 'A') AS a
                FROM seasons season
                JOIN season_clubs season_club ON season_club.season_id = season.id
                JOIN clubs club ON club.id = season_club.club_id
                LEFT JOIN player_registrations registration
                  ON registration.club_id = club.id AND registration.season_id = season.id
                 AND registration.registration_status = 'ACTIVE'
                LEFT JOIN player_reconciliation_results result
                  ON result.player_id = registration.player_id AND result.season_id = season.id
                LEFT JOIN players player ON player.id = result.player_id AND player.active = TRUE
                LEFT JOIN player_roles role
                  ON role.player_id = result.player_id AND role.season_id = season.id AND role.source = 'FANTACALCIO'
                WHERE season.code = :season
                GROUP BY club.id, club.name
                ORDER BY club.name
                """).param("season", season).query((rs, rowNum) -> new CatalogClubSummary(
                        rs.getLong("id"), rs.getString("name"), rs.getInt("total_players"),
                        rs.getInt("p"), rs.getInt("d"), rs.getInt("c"), rs.getInt("a"))).list();
    }

    private CatalogRoleSummary summarizeRoles(List<OperationalPlayer> players) {
        return new CatalogRoleSummary(countRole(players, "P"), countRole(players, "D"),
                countRole(players, "C"), countRole(players, "A"));
    }

    private int countRole(List<OperationalPlayer> players, String role) {
        return (int) players.stream().filter(player -> role.equals(player.role())).count();
    }

    private CatalogQualityIssue playerIssue(String code, String message, OperationalPlayer player) {
        return issue("ERROR", code, message, player.playerId(), player.fantacalcioExternalId(),
                player.name(), player.club(), Map.of());
    }

    private CatalogQualityIssue issue(String severity, String code, String message, Long playerId,
            String fantacalcioExternalId, String playerName, String club, Map<String, Object> details) {
        return new CatalogQualityIssue(severity, code, message, playerId, fantacalcioExternalId,
                playerName, club, Map.copyOf(details));
    }

    private int countSeverity(List<CatalogQualityIssue> issues, String severity) {
        return (int) issues.stream().filter(issue -> severity.equals(issue.severity())).count();
    }

    private List<String> messages(List<CatalogQualityIssue> issues, String severity) {
        return issues.stream().filter(issue -> severity.equals(issue.severity()))
                .map(CatalogQualityIssue::message).toList();
    }

    private void requireSeason(String season) {
        boolean exists = jdbcClient.sql("SELECT EXISTS (SELECT 1 FROM seasons WHERE code = :season)")
                .param("season", season).query(Boolean.class).single();
        if (!exists) {
            throw new ResourceNotFoundException("Stagione non trovata: " + season);
        }
    }

    private int countPendingTransfers(String season) {
        return jdbcClient.sql("""
                SELECT COUNT(*) FROM player_transfer_claims claim
                JOIN seasons season ON season.id = claim.season_id
                WHERE season.code = :season
                  AND claim.verification_status IN ('DETECTED', 'PENDING_VERIFICATION')
                """).param("season", season).query(Integer.class).single();
    }

    private int countOfficialExits(String season) {
        return jdbcClient.sql("""
                SELECT COUNT(*) FROM player_registration_events event
                JOIN seasons season ON season.id = event.season_id
                WHERE season.code = :season AND event.event_type = 'LEFT_SERIE_A'
                """).param("season", season).query(Integer.class).single();
    }

    private int countMissingFromFantacalcio(String season, long fantacalcioBatchId) {
        return jdbcClient.sql("""
                SELECT COUNT(DISTINCT result.player_id)
                FROM player_reconciliation_results result
                JOIN seasons season ON season.id = result.season_id
                JOIN players player ON player.id = result.player_id AND player.active = TRUE
                JOIN player_source_mappings mapping
                  ON mapping.player_id = player.id AND mapping.source = 'FANTACALCIO'
                 AND mapping.external_id IS NOT NULL AND mapping.verified = TRUE
                WHERE season.code = :season AND NOT EXISTS (
                    SELECT 1 FROM player_import_rows import_row
                    WHERE import_row.batch_id = :batchId AND import_row.valid = TRUE
                      AND import_row.external_id = mapping.external_id)
                """).param("season", season).param("batchId", fantacalcioBatchId)
                .query(Integer.class).single();
    }

    private BigDecimal calculateQuality(int confirmed, int total) {
        if (total <= 0) {
            return BigDecimal.ZERO.setScale(2, RoundingMode.HALF_UP);
        }
        return BigDecimal.valueOf(confirmed).multiply(BigDecimal.valueOf(100))
                .divide(BigDecimal.valueOf(total), 2, RoundingMode.HALF_UP);
    }

    private long requireLatestBatch(String season, String source) {
        Long id = optionalLatestBatch(season, source);
        if (id == null) {
            throw new ResourceNotFoundException("Nessun batch COMPLETED " + source + " per " + season);
        }
        return id;
    }

    private Long optionalLatestBatch(String season, String source) {
        return jdbcClient.sql("""
                SELECT batch.id FROM import_batches batch
                JOIN seasons season ON season.id = batch.season_id
                WHERE season.code = :season AND batch.source = :source AND batch.status = 'COMPLETED'
                  AND LOWER(batch.original_filename) NOT LIKE '%test%'
                ORDER BY COALESCE(batch.source_observed_at, batch.completed_at, batch.started_at) DESC,
                         batch.id DESC LIMIT 1
                """).param("season", season).param("source", source)
                .query(Long.class).optional().orElse(null);
    }

    record QualityReport(
            int totalOperationalPlayers,
            int playersWithoutFantacalcioRole,
            int playersWithoutActiveRegistration,

    @Transactional
    public TransferDetectionResponse detectLatest(

    ) {

        long fantacalcioBatchId =

                );

    @Transactional
    public TransferDetectionResponse detect(
            long fantacalcioBatchId,
            Long gazzettaBatchId,
            Long fantamarktBatchId
    ) {

        BatchInfo fantacalcio =
                batchInfo(
                        fantacalcioBatchId,
                        "FANTACALCIO"
                );

        if (gazzettaBatchId != null) {
            ensureSameSeason(
                    fantacalcio,
                    batchInfo(
                            gazzettaBatchId,
                            "GAZZETTA"
                    )
            );
        }

        if (fantamarktBatchId != null) {
            ensureSameSeason(
                    fantacalcio,
                    batchInfo(
                            fantamarktBatchId,
                            "FANTAMARKT"
                    )
            );
        }

        List<TransferCandidate> candidates =
                loadClubChangeCandidates(
                        fantacalcio
                );

        List<SerieAExitCandidate> exitCandidates =
                loadSerieAExitCandidates(
                        fantacalcio
                );

        int createdClaims = 0;
        int corroboratedClaims = 0;
        int existingOpenClaims = 0;
        int staleClaimsSkipped = 0;

        List<Long> createdClaimIds =
                new ArrayList<>();

        for (TransferCandidate candidate
                : candidates) {

            if (!lockAndMatchCurrentRegistration(candidate)) {
                staleClaimsSkipped++;
                continue;
            }

            if (hasNewerOfficialEvidence(
                    candidate,
                    fantacalcio.observedAt()
            )) {
                staleClaimsSkipped++;
                continue;
            }

            if (hasExistingOpenClaim(
                    candidate
            )) {
                existingOpenClaims++;
                continue;
            }

            boolean corroborated =
                    isCorroborated(
                            candidate,
                            gazzettaBatchId,
                            "GAZZETTA"
                    )
                    || isCorroborated(
                            candidate,
                            fantamarktBatchId,
                            "FANTAMARKT"
                    );

            String status =
                    corroborated
                            ? "PENDING_VERIFICATION"
                            : "DETECTED";

            Long claimId =
                    insertDetectedClaim(
                            candidate,
                            fantacalcio,
                            status
                    );

            createdClaims++;
            createdClaimIds.add(claimId);

            if (corroborated) {
                corroboratedClaims++;
            }
        }

        for (SerieAExitCandidate candidate : exitCandidates) {

            if (!lockAndMatchCurrentRegistration(candidate)) {
                staleClaimsSkipped++;
                continue;
            }

            if (hasNewerOfficialEvidence(
                    candidate.playerId(),
                    candidate.seasonId(),
                    fantacalcio.observedAt()
            )) {
                staleClaimsSkipped++;
                continue;
            }

            if (hasExistingOpenExitClaim(candidate)) {
                existingOpenClaims++;
                continue;
            }

            Long claimId =
                    insertDetectedExitClaim(
                            candidate,
                            fantacalcio
                    );

            createdClaims++;
            createdClaimIds.add(claimId);
        }

        int missingFromFantacalcio =
                countMissingFromFantacalcio(
                        fantacalcio
                );

        return new TransferDetectionResponse(
                fantacalcio.season(),
                fantacalcio.id(),
                gazzettaBatchId,
                fantamarktBatchId,
                candidates.size(),
                exitCandidates.size(),
                createdClaims,
                corroboratedClaims,
                existingOpenClaims,
                staleClaimsSkipped,
                missingFromFantacalcio,
                List.copyOf(createdClaimIds)
        );
    }

    private Long optionalLatestBatch(
            String season,
            String source
    ) {

        return jdbcClient.sql("""

                FROM import_batches batch

                JOIN seasons season
                    ON season.id = batch.season_id

                  AND batch.source = :source
                  AND batch.status = 'COMPLETED'

                ORDER BY
                    COALESCE(
                        batch.source_observed_at,
                        batch.completed_at,
                        batch.started_at
                    ) DESC,
                    batch.id DESC

                LIMIT 1
                """)

    private BatchInfo batchInfo(
            long batchId,
            String expectedSource
    ) {

        return jdbcClient.sql("""
                SELECT
                    batch.id,
                    season.id AS season_id,
                    season.code AS season,
                    batch.source,
                    batch.status,
                    batch.source_reference,

                    COALESCE(
                        batch.source_observed_at,
                        batch.completed_at,
                        batch.started_at
                    ) AS observed_at

                FROM import_batches batch

                JOIN seasons season
                    ON season.id = batch.season_id

                WHERE batch.id = :batchId
                """)
                .param("batchId", batchId)
                .query((resultSet, rowNumber) ->
                        new BatchInfo(
                                resultSet.getLong("id"),
                                resultSet.getLong(
                                        "season_id"
                                ),
                                resultSet.getString(
                                        "season"
                                ),
                                resultSet.getString(
                                        "source"
                                ),
                                resultSet.getString(
                                        "status"
                                ),
                                resultSet.getString(
                                        "source_reference"
                                ),
                                resultSet.getObject(
                                        "observed_at",
                                        OffsetDateTime.class
                                )
                        )
                )
                .optional()
                .map(batch -> {

                    if (!expectedSource.equals(
                            batch.source()
                    )) {
                        throw new ConflictException(
                                "Batch "
                                        + batchId
                                        + " atteso come "
                                        + expectedSource
                                        + " ma trovato "
                                        + batch.source()
                        );
                    }

                    if (!"COMPLETED".equals(
                            batch.status()
                    )) {
                        throw new ConflictException(
                                "Batch non completato: "
                                        + batchId
                        );
                    }

                    return batch;
                })
                .orElseThrow(() ->
                        new ResourceNotFoundException(
                                "Batch non trovato: "
                                        + batchId
                        )
                );
    }

                                batch.started_at

                            batch.id DESC
                        LIMIT 2
                        """)
                .param(
                        "season",
                        normalizedSeason
                )
                .query(
                        (resultSet, rowNumber) ->
                                new BatchSnapshot(
                                        resultSet.getLong("id"),
                                        resultSet.getObject(
                                                "observed_at",
                                                java.time.OffsetDateTime.class
                                        )
                                )
                )
                .list();

        /*
         * With no previous snapshot there is no evidence that an
         * entry is new. Never classify the whole current listone
         * as newly arrived.
         */
        if (batches.size() < 2) {
            return List.of();
        }

        BatchSnapshot current =
                batches.get(0);

        BatchSnapshot previous =
                batches.get(1);

        return jdbcClient.sql("""
                        SELECT DISTINCT
                            player.id AS player_id,
                            player.display_name AS player_name,
                            club.name AS club_name,
                            fantasy_role.fantasy_role,
                            season.code AS season_code,
                            current_row.external_id,

                            EXISTS (
                                SELECT 1
                                FROM player_registration_events event
                                WHERE event.player_id = player.id
                                  AND event.season_id = season.id
                                  AND event.event_type =
                                      'TRANSFER_IN_SERIE_A'
                            ) AS official_transfer_event_present

                        FROM player_import_rows current_row

                        JOIN import_batches current_batch
                            ON current_batch.id =
                               current_row.batch_id

                        JOIN seasons season
                            ON season.id =
                               current_batch.season_id

                        JOIN player_source_mappings mapping
                            ON mapping.source =
                               'FANTACALCIO'
                           AND mapping.external_id =
                               current_row.external_id
                           AND mapping.verified = TRUE

                        JOIN players player
                            ON player.id = mapping.player_id
                           AND player.active = TRUE

                        JOIN player_registrations registration
                            ON registration.player_id =
                               player.id
                           AND registration.season_id =
                               season.id
                           AND registration.registration_status =
                               'ACTIVE'

                        JOIN season_clubs membership
                            ON membership.season_id =
                               season.id
                           AND membership.club_id =
                               registration.club_id

                        JOIN clubs club
                            ON club.id =
                               registration.club_id

                        JOIN player_roles fantasy_role
                            ON fantasy_role.player_id =
                               player.id
                           AND fantasy_role.season_id =
                               season.id
                           AND fantasy_role.source =
                               'FANTACALCIO'

                        WHERE current_row.batch_id =
                              :currentBatchId

                          AND current_row.valid = TRUE

                          AND current_row.external_id
                              IS NOT NULL

                          AND BTRIM(
                              current_row.external_id
                          ) <> ''

                          AND LOWER(
                              COALESCE(
                                  current_row.raw_data
                                      ->> 'out_of_game',
                                  'false'
                              )
                          ) <> 'true'

                          AND NOT EXISTS (
                              SELECT 1
                              FROM player_import_rows previous_row
                              WHERE previous_row.batch_id =
                                    :previousBatchId
                                AND previous_row.valid = TRUE
                                AND previous_row.external_id =
                                    current_row.external_id
                          )

                          AND NOT EXISTS (
                              SELECT 1
                              FROM foreign_player_scouting_reports report
                              WHERE report.player_id =
                                    player.id
                                AND report.season_id =
                                    season.id
                          )

                          AND (
                              CAST(:role AS VARCHAR) IS NULL
                              OR fantasy_role.fantasy_role =
                                  CAST(:role AS VARCHAR)
                          )

                          AND (
                              CAST(:club AS VARCHAR) IS NULL
                              OR LOWER(club.name) =
                                  LOWER(
                                      CAST(
                                          :club AS VARCHAR
                                      )
                                  )
                              OR LOWER(club.short_name) =
                                  LOWER(
                                      CAST(
                                          :club AS VARCHAR
                                      )
                                  )
                              OR LOWER(club.slug) =
                                  LOWER(
                                      CAST(
                                          :club AS VARCHAR
                                      )
                                  )
                          )

                        ORDER BY
                            player.id ASC
                        """)
                .param(
                        "currentBatchId",
                        current.id()
                )
                .param(
                        "previousBatchId",
                        previous.id()
                )
                .param(
                        "role",
                        normalizedRole
                )
                .param(
                        "club",
                        normalizedClub
                )
                .query(
                        (resultSet, rowNumber) ->
                                new B70NewListoneArrivalResponse(
                                        resultSet.getLong(
                                                "player_id"
                                        ),
                                        resultSet.getString(
                                                "player_name"
                                        ),
                                        resultSet.getString(
                                                "club_name"
                                        ),
                                        resultSet.getString(
                                                "fantasy_role"
                                        ),
                                        resultSet.getString(
                                                "season_code"
                                        ),
                                        resultSet.getString(
                                                "external_id"
                                        ),
                                        current.id(),
                                        previous.id(),
                                        current.observedAt(),
                                        "NEW_FANTACALCIO_LISTONE_ENTRY",
                                        resultSet.getBoolean(
                                                "official_transfer_event_present"
                                        )
                                )
                )
                .list();
    }

    private String normalizeRole(
            String role
    ) {

        String normalized =
                normalizeOptionalText(role);

        if (normalized == null) {
            return null;
        }

        normalized =
                normalized.toUpperCase(
                        Locale.ROOT
                );

        if (!ROLES.contains(normalized)) {

            throw new BadRequestException(
                    "Ruolo non valido: "
                            + role
            );
        }

        return normalized;
    }

    private String requireText(
            String value,
            String message
    ) {

        if (
            value == null
            || value.isBlank()
        ) {

            throw new BadRequestException(
                    message
            );
        }

        return value.trim();
    }

    private String normalizeOptionalText(
            String value
    ) {

        if (
            value == null
            || value.isBlank()
        ) {
            return null;
        }

        return value.trim();
    }

    private record BatchSnapshot(
            Long id,
            java.time.OffsetDateTime observedAt
    ) {
    }
